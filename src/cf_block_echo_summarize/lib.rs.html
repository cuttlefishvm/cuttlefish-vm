1//! Example proc-block: summarizes the beginning of a local file.
2//!
3//! This is a fixture as much as a sample. The host's integration tests build it
4//! to `wasm32-unknown-unknown` and drive it through a real wasm boundary, which
5//! is what makes those tests exercise the actual ABI rather than a mock of it.
6//!
7//! Note what this block never does: hold the file. It sees a handle and a
8//! length, then decides for itself how much to pull into its own memory. A block
9//! wanting the whole file would loop on `Slice` from `next_offset`, and its
10//! memory ceiling would still be `WINDOW` rather than the file's size.
11
12use cuttlefish_sdk::{export_block, Block, Command, Event, MediaKind, Signature, TokenAction};
13
14/// Build the summarization prompt.
15///
16/// The obvious phrasing — "Summarize the following:" then the text — reads to a
17/// small model as an assertion to evaluate rather than content to condense.
18/// Given a one-line document, llama3.2:1b answered "There is no information
19/// about cuttlefish delegating jobs to local models in my knowledge base",
20/// which looks like a broken pipeline and is not: the text arrived intact.
21///
22/// Delimiting the document and saying explicitly not to judge it fixes that
23/// with the same model. Prompt shape is part of a block's behaviour, not an
24/// afterthought.
25fn summarize_prompt(text: &str) -> String {
26    format!(
27        "Summarize the document between the markers in one or two sentences. \
28         Do not comment on whether it is true.\n\n<<<DOCUMENT\n{}\nDOCUMENT>>>",
29        text.trim()
30    )
31}
32
33/// How much of a file this block is willing to hold at once.
34///
35/// The particular value is arbitrary; the point is that a bound exists. Guest
36/// memory is governed by this rather than by the size of the input.
37const WINDOW: u64 = 64 * 1024;
38
39#[derive(Default)]
40struct EchoSummarize {
41    path: String,
42    /// Set from the job input purely so the host's early-stop path has a block
43    /// that exercises it. A real block would decide from the content it sees.
44    stop_after_first: bool,
45    seen_tokens: u32,
46}
47
48impl Block for EchoSummarize {
49    fn signature() -> Signature {
50        // Takes a path, produces that path alongside a summary. Stated
51        // concretely rather than as `json` so a pipeline seam involving this
52        // block is actually checked — a `json` seam typechecks unconditionally,
53        // which is the same as not checking it.
54        Signature {
55            input: "{path: text}".parse().expect("a literal type"),
56            output: "{path: text, summary: text}"
57                .parse()
58                .expect("a literal type"),
59        }
60    }
61
62    fn start(&mut self, input: serde_json::Value) -> Command {
63        self.stop_after_first = input
64            .get("stop_after_first")
65            .and_then(|v| v.as_bool())
66            .unwrap_or(false);
67
68        match input.get("path").and_then(|v| v.as_str()) {
69            Some(path) => {
70                self.path = path.to_string();
71                Command::Open {
72                    path: path.to_string(),
73                }
74            }
75            // Fail rather than panic: a panic becomes an opaque wasm trap, while
76            // this reaches the caller as a code they can act on.
77            None => Command::Fail {
78                code: "schema_validation_failed".into(),
79                message: "input must have a string `path` field".into(),
80            },
81        }
82    }
83
84    fn step(&mut self, event: Event) -> Command {
85        match event {
86            // Branch on what the host actually found, rather than assuming
87            // text. A PDF gets read through its text layer when it has one, and
88            // rendered for a vision model when it does not — which is the
89            // difference between summarizing a scan and summarizing nothing.
90            Event::Opened { handle, len, kind } => match kind {
91                MediaKind::Document {
92                    has_text_layer: true,
93                    ..
94                } => Command::PageText { handle, page: 0 },
95                MediaKind::Document { .. } => Command::PageImage { handle, page: 0 },
96                MediaKind::Image { .. } => Command::Infer {
97                    prompt: "Describe this image.".into(),
98                    max_tokens: 128,
99                    images: vec![handle],
100                },
101                MediaKind::Text => Command::Slice {
102                    handle,
103                    offset: 0,
104                    len: len.min(WINDOW),
105                },
106                MediaKind::Binary => Command::Fail {
107                    code: "unsupported".into(),
108                    message: "this block reads text, images, and documents".into(),
109                },
110            },
111            Event::Sliced { text, .. } => Command::Infer {
112                prompt: summarize_prompt(&text),
113                max_tokens: 128,
114                images: Vec::new(),
115            },
116            Event::PageTexted { text } => Command::Infer {
117                prompt: summarize_prompt(&text),
118                max_tokens: 128,
119                images: Vec::new(),
120            },
121            // A rendered page is an image handle like any other.
122            Event::PageImaged { handle, .. } => Command::Infer {
123                prompt: "Read this page and summarize it.".into(),
124                max_tokens: 128,
125                images: vec![handle],
126            },
127            Event::InferDone { text, .. } => Command::Done {
128                result: serde_json::json!({ "path": self.path, "summary": text }),
129            },
130            Event::SlicedBytes { .. } | Event::Emitted => Command::Fail {
131                code: "unexpected_event".into(),
132                message: "this block requests neither raw bytes nor progress".into(),
133            },
134        }
135    }
136
137    fn on_token(&mut self, _token: &str) -> TokenAction {
138        self.seen_tokens += 1;
139        if self.stop_after_first {
140            TokenAction::Stop
141        } else {
142            TokenAction::Continue
143        }
144    }
145}
146
147export_block!(EchoSummarize);
