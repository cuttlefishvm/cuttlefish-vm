1//! Host-side image transforms: the operations that need a decoder.
2//!
3//! The dividing line this module sits on is the same one `pdf-render` sits
4//! on. Reading what an image *says about itself* — dimensions, EXIF, chunk
5//! structure — is header parsing, cheap, and available to any script in the
6//! guest with no feature flag and no round trip. Reading what an image
7//! *looks like* means decoding pixels, which is megabytes of codec that has
8//! no business in a wasm guest shipped to every user.
9//!
10//! So this is behind the `image-ops` feature, off by default. A host built
11//! without it rejects [`cuttlefish_abi::Command::ImageOp`] with a message
12//! saying exactly that, rather than silently returning the original image —
13//! which would look like a working pipeline producing quietly wrong results.
14
15/// Apply one operation, returning re-encoded PNG bytes.
16///
17/// PNG on the way out regardless of what came in: it is lossless, so a crop
18/// fed to a vision model is not also silently re-compressed, and it is what
19/// `PageImage` already produces, so every image handle in the system has one
20/// encoding rather than two.
21#[cfg(feature = "image-ops")]
22pub fn apply(bytes: &[u8], op: &cuttlefish_abi::ImageOperation) -> Result<Vec<u8>, anyhow::Error> {
23    use cuttlefish_abi::ImageOperation;
24    use image::GenericImageView as _;
25
26    let decoded =
27        image::load_from_memory(bytes).map_err(|e| anyhow::anyhow!("decoding the image: {e}"))?;
28
29    let out = match op {
30        ImageOperation::Resize {
31            max_width,
32            max_height,
33        } => {
34            if *max_width == 0 || *max_height == 0 {
35                anyhow::bail!("resize bounds must both be non-zero");
36            }
37            // `thumbnail` preserves aspect ratio but *will* enlarge a small
38            // image to fill the box, so the bounds are clamped to the
39            // original first. "Fit within these bounds" must never mean
40            // upscale: invented detail is detail a vision model then
41            // reasons about as though it were real.
42            let (w, h) = decoded.dimensions();
43            decoded.thumbnail((*max_width).min(w), (*max_height).min(h))
44        }
45        ImageOperation::Crop {
46            x,
47            y,
48            width,
49            height,
50        } => {
51            if *width == 0 || *height == 0 {
52                anyhow::bail!("crop width and height must both be non-zero");
53            }
54            let (w, h) = decoded.dimensions();
55            // Refuse rather than clamp. A crop that silently returns a
56            // different region than asked for is worse than an error: the
57            // caller reasons about coordinates it did not get.
58            if x.saturating_add(*width) > w || y.saturating_add(*height) > h {
59                anyhow::bail!(
60                    "crop region {width}x{height}+{x}+{y} does not fit inside the \
61                     image's {w}x{h}"
62                );
63            }
64            decoded.crop_imm(*x, *y, *width, *height)
65        }
66    };
67
68    let mut encoded = std::io::Cursor::new(Vec::new());
69    out.write_to(&mut encoded, image::ImageFormat::Png)
70        .map_err(|e| anyhow::anyhow!("re-encoding the image: {e}"))?;
71    Ok(encoded.into_inner())
72}

74/// Without the feature, say so plainly.
75///
76/// Not a silent pass-through of the original bytes: that would produce a
77/// pipeline that appears to work while every crop and resize quietly did
78/// nothing, and the caller would have no way to tell.
79#[cfg(not(feature = "image-ops"))]
80pub fn apply(
81    _bytes: &[u8],
82    _op: &cuttlefish_abi::ImageOperation,
83) -> Result<Vec<u8>, anyhow::Error> {
84    anyhow::bail!(
85        "this cuttlefish was built without the `image-ops` feature, so it cannot decode \
86         images — rebuild with `--features image-ops`. Image *metadata* (dimensions, EXIF, \
87         chunk structure) needs no feature and is available to any script via the \
88         `dimensions`/`exif`/`png_chunks` builtins."
89    )
90}

92#[cfg(all(test, feature = "image-ops"))]
93mod tests {
94    use super::*;
95    use cuttlefish_abi::ImageOperation;
96
97    fn png(width: u32, height: u32) -> Vec<u8> {
98        let img =
99            image::RgbImage::from_fn(width, height, |x, _| image::Rgb([(x % 256) as u8, 0, 0]));
100        let mut out = std::io::Cursor::new(Vec::new());
101        image::DynamicImage::ImageRgb8(img)
102            .write_to(&mut out, image::ImageFormat::Png)
103            .unwrap();
104        out.into_inner()
105    }
106
107    fn size_of(bytes: &[u8]) -> (u32, u32) {
108        use image::GenericImageView as _;
109        image::load_from_memory(bytes).unwrap().dimensions()
110    }
111
112    #[test]
113    fn resize_fits_within_bounds_and_keeps_the_aspect_ratio() {
114        let out = apply(
115            &png(800, 400),
116            &ImageOperation::Resize {
117                max_width: 100,
118                max_height: 100,
119            },
120        )
121        .unwrap();
122        // 2:1 fitted into a square box is 100x50, not 100x100 — a stretched
123        // image would make a vision model describe the wrong shapes.
124        assert_eq!(size_of(&out), (100, 50));
125    }
126
127    #[test]
128    fn resize_never_enlarges() {
129        // Upscaling invents detail the model would then reason about.
130        let out = apply(
131            &png(10, 10),
132            &ImageOperation::Resize {
133                max_width: 500,
134                max_height: 500,
135            },
136        )
137        .unwrap();
138        assert_eq!(size_of(&out), (10, 10));
139    }
140
141    #[test]
142    fn crop_returns_exactly_the_requested_region() {
143        let out = apply(
144            &png(100, 100),
145            &ImageOperation::Crop {
146                x: 10,
147                y: 20,
148                width: 30,
149                height: 40,
150            },
151        )
152        .unwrap();
153        assert_eq!(size_of(&out), (30, 40));
154    }
155
156    #[test]
157    fn a_crop_outside_the_image_is_refused_rather_than_clamped() {
158        // Silently returning a different region than asked for is worse
159        // than an error: the caller reasons about coordinates it never got.
160        let err = apply(
161            &png(50, 50),
162            &ImageOperation::Crop {
163                x: 40,
164                y: 40,
165                width: 30,
166                height: 30,
167            },
168        )
169        .expect_err("an out-of-bounds crop must be refused");
170        assert!(err.to_string().contains("does not fit"), "{err}");
171    }
172
173    #[test]
174    fn zero_sized_operations_are_refused() {
175        assert!(apply(
176            &png(10, 10),
177            &ImageOperation::Resize {
178                max_width: 0,
179                max_height: 10
180            }
181        )
182        .is_err());
183        assert!(apply(
184            &png(10, 10),
185            &ImageOperation::Crop {
186                x: 0,
187                y: 0,
188                width: 0,
189                height: 5
190            }
191        )
192        .is_err());
193    }
194
195    #[test]
196    fn undecodable_bytes_error_rather_than_panic() {
197        let err = apply(
198            b"not an image at all",
199            &ImageOperation::Resize {
200                max_width: 10,
201                max_height: 10,
202            },
203        )
204        .expect_err("garbage must not decode");
205        assert!(err.to_string().contains("decoding"), "{err}");
206    }
