1//! Typechecking a node graph: topological order, fan-in composition via
2//! `InputExpr`, cycle rejection (unless marked `repeat_until`), and
3//! branch-exclusivity analysis for conditional dispatch.
4//!
5//! See docs/superpowers/specs/2026-08-03-dag-core-design.md for the full
6//! design. This module's `check_graph` is the graph-shaped analogue of
7//! `crate::pipeline::check` — same idea (typecheck seams before anything
8//! runs), different shape (a graph of named nodes instead of a `Vec`).
9
10use crate::pipeline::{read_stage_signature, PipelineError, ResolvedInput};
11use cuttlefish_abi::Ty;
12use cuttlefish_core::graph::{Branches, InputExpr, NodeGraph};
13use std::collections::{BTreeMap, HashMap, HashSet};
14use wasmtime::Engine;

16/// One checked node, ready to execute.
17///
18/// `Clone` because a later task's `resume_job` rebuilds a `JobSpec` from an
19/// `Arc<Vec<CheckedNode>>` held in `AppState` — every field type here
20/// (`Signature`, `ArtifactKind`, `InputExpr`) is already `Clone`.
21#[derive(Clone)]
22pub struct CheckedNode {
23    /// The node's name, as written in `nodes = {...}`.
24    pub name: String,
25    /// Block or bundle.
26    pub kind: crate::catalog::ArtifactKind,
27    /// The exact `name@version` this node resolved to, if it came from the
28    /// catalog.
29    pub resolved: Option<String>,
30    /// The compiled module (block) or `.cfbundle` (bundle) bytes.
31    pub module_bytes: Vec<u8>,
32    /// What it declared.
33    pub signature: cuttlefish_abi::Signature,
34    /// What feeds this node, if anything.
35    pub input: Option<InputExpr>,
36    /// Bounded-loop marker, if this node re-runs on its own output.
37    pub repeat_until: Option<String>,
38    /// Iteration bound, required alongside `repeat_until`.
39    pub max_iterations: Option<u32>,
40    /// Threaded straight from `ResolvedInput::script`/`Stage::script` — see
41    /// `pipeline.rs`. `Some` only for a `Script`-kind node.
42    pub script: Option<String>,
43    /// The fan-out manifest this node runs over, if any — see
44    /// [`cuttlefish_core::graph::Node::over`]. When set, this node runs its
45    /// block once per manifest line and presents
46    /// [`fanout_collection_ty`] downstream rather than its block's own
47    /// declared output.
48    pub over: Option<std::path::PathBuf>,
49    /// For a fan-out node only: what its block declared as the output of
50    /// *one item*, before [`fanout_collection_ty`] replaced `signature.output`
51    /// for downstream typing.
52    ///
53    /// Both are needed and they are not the same type. Downstream nodes
54    /// consume the collection, so `signature.output` must describe that; but
55    /// each individual item's result still has to be validated against what
56    /// the block actually promised to produce, which is this. Collapsing them
57    /// into one field means every item gets checked against the collection
58    /// record and fails.
59    pub item_output: Option<cuttlefish_abi::Ty>,
60}
61
62/// What a fan-out node presents to the nodes downstream of it.
63///
64/// Deliberately *not* the block's own declared output: downstream consumes
65/// the collection of every item's result, not any one item's. The counts are
66/// [`Ty::Json`] because [`Ty`] has no number variant.
67pub fn fanout_collection_ty() -> Ty {
68    Ty::Record(BTreeMap::from([
69        ("results_path".to_string(), Ty::Text),
70        ("failures_path".to_string(), Ty::Text),
71        ("succeeded".to_string(), Ty::Json),
72        ("failed".to_string(), Ty::Json),
73    ]))
74}

76/// A whole graph, typechecked and topologically ordered.
77pub struct CheckedGraph {
78    /// In topological order — safe to execute front-to-back, threading
79    /// `outputs` forward, per the spec's execution-semantics section.
80    pub nodes: Vec<CheckedNode>,
81    /// Which nodes are exclusive to which branch label, keyed by the
82    /// branching node's name — see "skip propagation" in `check_graph`.
83    pub exclusive_to: HashMap<String, BranchExclusivity>,
84}
85
86/// Which branch decision + label a node is exclusive to — a node is only
87/// executed when this decision's chosen route matches `label`. Carrying
88/// `decision` (not just `label`) is what lets two independent `branches`
89/// decisions that happen to reuse the same label string coexist without
90/// being confused for a conflict.
91#[derive(Debug, Clone, PartialEq, Eq)]
92pub struct BranchExclusivity {
93    /// The branching decision (key in `branches.decisions`) this exclusivity
94    /// belongs to.
95    pub decision: String,
96    /// The label within that decision.
97    pub label: String,
98}

100/// Why a graph was rejected.
101#[derive(Debug, thiserror::Error)]
102pub enum DagError {
103    /// A per-node signature lookup or seam check failed the same way a
104    /// linear pipeline's would.
105    #[error(transparent)]
106    Pipeline(#[from] PipelineError),
107    /// A node's `in` expression names a node that doesn't exist.
108    #[error("node `{node}` references unknown node `{referenced}`")]
109    UnknownReference {
110        /// The node whose `in` expression has the bad reference.
111        node: String,
112        /// The name it referenced.
113        referenced: String,
114    },
115    /// A node has an inbound edge that would form a cycle without an
116    /// explicit `repeat_until` marker.
117    #[error(
118        "node `{node}` has an inbound edge that would form a cycle with no \
119         repeat_until marker — add one (with max_iterations) if this loop is intentional"
120    )]
121    UnmarkedCycle {
122        /// The node that could not be ordered.
123        node: String,
124    },
125    /// A node's input mixes outputs from two mutually-exclusive branch labels
126    /// of the same decision.
127    #[error(
128        "node `{node}`'s input mixes output from branch label `{label_a}` and \
129         label `{label_b}` of the same `branches.{decision}` decision — a node \
130         cannot depend on more than one mutually-exclusive branch outcome at once"
131    )]
132    ConflictingBranchFanIn {
133        /// The node whose input mixes two labels.
134        node: String,
135        /// The branching decision both labels belong to.
136        decision: String,
137        /// The first label found.
138        label_a: String,
139        /// The second, conflicting label.
140        label_b: String,
141    },
142    /// A node's declared input doesn't accept what its `in` expression
143    /// produces.
144    #[error("node `{consumer}` needs {expected}, but `{producer}` produces {produced}")]
145    SeamMismatch {
146        /// What produced the mismatched value.
147        producer: String,
148        /// What it produces.
149        produced: String,
150        /// The node that needed something else.
151        consumer: String,
152        /// What it needs.
153        expected: String,
154    },
155    /// The graph had no nodes.
156    #[error("a graph needs at least one node")]
157    Empty,
158}

160/// Typecheck a graph. `resolved` must already contain one entry per node in
161/// `graph.nodes`, in the same order — building that mapping (via
162/// `pipeline::resolve_and_load`, one call per node) is the caller's job
163/// (a later task), same division of responsibility `pipeline::check` already has.
164pub fn check_graph(
165    engine: &Engine,
166    graph: &NodeGraph,
167    branches: &Branches,
168    resolved: &HashMap<String, ResolvedInput>,
169) -> Result<CheckedGraph, DagError> {
170    if graph.nodes.is_empty() {
171        return Err(DagError::Empty);
172    }
173
174    // 1. Read every node's signature up front — needed before topological
175    //    evaluation since InputExpr composition needs to know each
176    //    referenced node's *output* type, and a node can be referenced
177    //    before it's "current" in visit order.
178    let mut signatures = HashMap::new();
179    let mut item_outputs: HashMap<String, cuttlefish_abi::Ty> = HashMap::new();
180    for (name, node) in &graph.nodes {
181        let input = resolved.get(name).expect("caller resolved every node");
182        let mut signature = read_stage_signature(engine, input)?;
183        if node.over.is_some() {
184            // Keep what the block promised for one item — the substitution
185            // below is about downstream typing only, and each item's result
186            // still has to be checked against this at runtime.
187            item_outputs.insert(name.clone(), signature.output.clone());
188            // A fan-out node's block declares the shape of *one item's*
189            // result, but downstream nodes consume the collection of all of
190            // them. Substituting here — where per-node signatures are first
191            // collected — is what makes every seam check downstream correct
192            // with no further changes, since `evaluate_expr_ty` and each
193            // `assignable_to` comparison read from this same map. The
194            // block's own declared output is still used, per item, to
195            // validate what each run produced.
196            signature.output = fanout_collection_ty();
197        }
198        signatures.insert(name.clone(), signature);
199    }
200
201    // 2. Topological sort with cycle detection. An edge node -> referenced
202    //    is only legal going "backward" (referenced already visited) unless
203    //    `node` declares repeat_until, in which case a self-edge is exactly
204    //    what's expected and not an error.
205    let order = topological_order(graph)?;
206
207    // 3. Branch-exclusivity: for each `branches` decision, walk forward from
208    //    each label's target, marking every node whose InputExpr needs that
209    //    target (transitively) as exclusive to that label. A node needing
210    //    two labels of the *same* decision is a build-time error.
211    let exclusive_to = compute_branch_exclusivity(graph, branches)?;
212
213    // 4. For each node in topological order, evaluate its InputExpr into a
214    //    Ty (composing referenced nodes' output types) and check
215    //    assignable_to against its own declared input.
216    let mut nodes = Vec::with_capacity(order.len());
217    for name in &order {
218        let node = graph
219            .get(name)
220            .expect("topological_order only returns known nodes");
221        let input_resolved = resolved.get(name).expect("caller resolved every node");
222        let signature = signatures.get(name).unwrap().clone();
223
224        if let Some(expr) = &node.input {
225            let produced = evaluate_expr_ty(expr, &signatures, graph)?;
226            if !produced.assignable_to(&signature.input) {
227                let (producer, produced_str) = describe_expr(expr, &signatures);
228                return Err(DagError::SeamMismatch {
229                    producer,
230                    produced: produced_str,
231                    consumer: name.clone(),
232                    expected: signature.input.to_string(),
233                });
234            }
235        }
236
237        nodes.push(CheckedNode {
238            name: name.clone(),
239            kind: input_resolved.kind,
240            resolved: input_resolved.resolved.clone(),
241            module_bytes: input_resolved.bytes.clone(),
242            signature,
243            input: node.input.clone(),
244            repeat_until: node.repeat_until.clone(),
245            max_iterations: node.max_iterations,
246            script: input_resolved.script.clone(),
247            over: node.over.clone(),
248            item_output: item_outputs.get(name).cloned(),
249        });
250    }
251
252    Ok(CheckedGraph {
253        nodes,
254        exclusive_to,
255    })
256}

258/// A stable fingerprint of a checked graph's shape and contents — every
259/// node's name and declared signature, joined and hashed. Two graphs with
260/// the same fingerprint are the same, for resume-safety purposes; this
261/// isn't a security boundary, just a "did the loaded spec actually change"
262/// guard, so a simple SHA-256 (already a workspace dependency, same crate
263/// the catalog's own content-hashing uses) is all this needs.
264pub fn graph_fingerprint(nodes: &[CheckedNode]) -> String {
265    use sha2::{Digest, Sha256};
266    let mut hasher = Sha256::new();
267    for node in nodes {
268        hash_length_prefixed(&mut hasher, node.name.as_bytes());
269        hash_length_prefixed(&mut hasher, node.signature.to_string().as_bytes());
270        // A node fanning out over a different manifest is a different node
271        // for resume purposes even with an identical signature: its recorded
272        // item indices refer to inputs from the old manifest. Without this,
273        // repointing `over` would resume against checkpoints computed from
274        // entirely different data.
275        hash_length_prefixed(
276            &mut hasher,
277            node.over
278                .as_ref()
279                .map(|p| p.as_os_str().as_encoded_bytes())
280                .unwrap_or(b""),
281        );
282    }
283    format!("{:x}", hasher.finalize())
284}

286/// Feed `bytes` into `hasher` prefixed with its own length (as a fixed
287/// 8-byte big-endian `u64`), rather than delimiting with a fixed byte —
288/// delimiter-joining is only injective if the delimiter can never appear
289/// inside the content itself, which isn't guaranteed here (a wasm block's
290/// declared signature can contain arbitrary field-name text, including, in
291/// principle, an embedded NUL byte decoded from its `cf_signature` export's
292/// JSON). Length-prefixing has no such assumption.
293fn hash_length_prefixed(hasher: &mut sha2::Sha256, bytes: &[u8]) {
294    use sha2::Digest;
295    hasher.update((bytes.len() as u64).to_be_bytes());
296    hasher.update(bytes);
297}

299/// Kahn's algorithm, with the repeat_until exception: an edge from `node`
300/// back to itself (or forming a cycle) is only legal when `node` declares
301/// `repeat_until` — everything else with an unresolved inbound edge after
302/// the sort terminates is an undeclared cycle.
303fn topological_order(graph: &NodeGraph) -> Result<Vec<String>, DagError> {
304    let mut deps: HashMap<&str, HashSet<&str>> = HashMap::new();
305    for (name, node) in &graph.nodes {
306        deps.entry(name).or_default();
307        if let Some(expr) = &node.input {
308            for referenced in referenced_nodes(expr) {
309                if graph.get(referenced).is_none() {
310                    return Err(DagError::UnknownReference {
311                        node: name.clone(),
312                        referenced: referenced.to_string(),
313                    });
314                }
315                // A node's own repeat_until self-reference is not a real
316                // dependency edge for ordering purposes — it re-runs on its
317                // own prior output, which the executor (a later task)
318                // special-cases, not the topological sort. A self-reference
319                // WITHOUT repeat_until is not this case — it's exactly the
320                // undeclared-cycle mistake the spec's `repeat_until`-as-sole-
321                // marker rule exists to catch, so it must still become a
322                // dependency edge (on itself) that can never be satisfied,
323                // tripping UnmarkedCycle below.
324                let is_marked_self_loop = referenced == name && node.repeat_until.is_some();
325                if !is_marked_self_loop {
326                    deps.entry(name).or_default().insert(referenced);
327                }
328            }
329        }
330    }
331
332    let mut order = Vec::new();
333    let mut remaining: HashMap<&str, HashSet<&str>> = deps.clone();
334    loop {
335        let ready: Vec<&str> = remaining
336            .iter()
337            .filter(|(_, d)| d.is_empty())
338            .map(|(n, _)| *n)
339            .collect();
340        if ready.is_empty() {
341            break;
342        }
343        let mut ready = ready;
344        ready.sort(); // deterministic order among independent nodes
345        for n in &ready {
346            order.push(n.to_string());
347            remaining.remove(n);
348        }
349        for deps in remaining.values_mut() {
350            for n in &ready {
351                deps.remove(n);
352            }
353        }
354    }
355
356    if order.len() != graph.nodes.len() {
357        let stuck = graph
358            .nodes
359            .iter()
360            .map(|(n, _)| n.as_str())
361            .find(|n| !order.contains(&n.to_string()))
362            .unwrap();
363        return Err(DagError::UnmarkedCycle {
364            node: stuck.to_string(),
365        });
366    }
367    Ok(order)
368}

370fn referenced_nodes(expr: &InputExpr) -> Vec<&str> {
371    match expr {
372        InputExpr::FromNode(n) => vec![n.as_str()],
373        InputExpr::Record(fields) => fields.values().flat_map(referenced_nodes).collect(),
374        InputExpr::List(items) => items.iter().flat_map(referenced_nodes).collect(),
375    }
376}

378/// Compose an `InputExpr` into the `Ty` it produces, by looking up each
379/// referenced node's declared output type.
380// `graph` is only threaded through the recursive calls today (not read at
381// any base case); kept as a parameter rather than dropped since a future
382// case is plausible to need it (e.g. validating a reference more richly
383// than `signatures` alone allows) and this is purely a lint suppression,
384// not a change in behavior.
385#[allow(clippy::only_used_in_recursion)]
386fn evaluate_expr_ty(
387    expr: &InputExpr,
388    signatures: &HashMap<String, cuttlefish_abi::Signature>,
389    graph: &NodeGraph,
390) -> Result<Ty, DagError> {
391    match expr {
392        InputExpr::FromNode(n) => Ok(signatures
393            .get(n)
394            .ok_or_else(|| DagError::UnknownReference {
395                node: "?".into(),
396                referenced: n.clone(),
397            })?
398            .output
399            .clone()),
400        InputExpr::Record(fields) => {
401            let mut out = BTreeMap::new();
402            for (k, v) in fields {
403                out.insert(k.clone(), evaluate_expr_ty(v, signatures, graph)?);
404            }
405            Ok(Ty::Record(out))
406        }
407        InputExpr::List(items) => {
408            // All list items must agree on a type for `Ty::List(T)` to mean
409            // anything; take the first and let assignable_to's own equality
410            // fall through to a SeamMismatch if a later one disagrees. (A
411            // more precise per-item error is a reasonable follow-up; this is
412            // the minimal correct behavior for v1.)
413            let first = items.first().ok_or_else(|| DagError::UnknownReference {
414                node: "?".into(),
415                referenced: "<empty list>".into(),
416            })?;
417            Ok(Ty::List(Box::new(evaluate_expr_ty(
418                first, signatures, graph,
419            )?)))
420        }
421    }
422}

424fn describe_expr(
425    expr: &InputExpr,
426    signatures: &HashMap<String, cuttlefish_abi::Signature>,
427) -> (String, String) {
428    match expr {
429        InputExpr::FromNode(n) => (
430            n.clone(),
431            signatures
432                .get(n)
433                .map(|s| s.output.to_string())
434                .unwrap_or_default(),
435        ),
436        InputExpr::Record(fields) => (
437            "<composite>".to_string(),
438            match same_node_repeated_across_every_field(fields) {
439                // The single most common way to reach for `Record` wrong:
440                // wanting one upstream node's whole output passed through
441                // to a downstream node whose input shape happens to match
442                // it, but writing `{ a = x.out; b = x.out; c = x.out; }`
443                // instead of the bare `in = x.out;` a straight pass-through
444                // actually needs. The wrapped form nests x's whole output
445                // under *each* field instead of using its fields directly,
446                // producing the double-nested type this message would
447                // otherwise show with no explanation. Caught here rather
448                // than left to be found by trial and error against a
449                // confusing type dump -- see the cuttlefish-build skill.
450                Some(node) => format!(
451                    "{expr:?} -- every field here maps to `{node}.out`; if you meant to pass \
452                     `{node}`'s whole output through unchanged, write `in = {node}.out;` with no \
453                     braces instead of wrapping it field by field"
454                ),
455                None => format!("{expr:?}"),
456            },
457        ),
458        other => ("<composite>".to_string(), format!("{other:?}")),
459    }
460}

462/// If every field of a `Record` maps to the same single node's whole
463/// output (`FromNode`), returns that node's name — see the call site in
464/// [`describe_expr`] for why this is worth detecting.
465fn same_node_repeated_across_every_field(fields: &BTreeMap<String, InputExpr>) -> Option<&str> {
466    let mut names = fields.values().map(|v| match v {
467        InputExpr::FromNode(n) => Some(n.as_str()),
468        _ => None,
469    });
470    let first = names.next()??;
471    names.all(|n| n == Some(first)).then_some(first)
472}

474/// Implements the spec's "Conditional dispatch and skipped nodes" rule
475/// exactly: a node is exclusive to label L if any node its InputExpr
476/// references is exclusive to L (transitively, from L's branch target). A
477/// node that would be exclusive to two labels of the *same* decision at
478/// once is a build-time error.
479fn compute_branch_exclusivity(
480    graph: &NodeGraph,
481    branches: &Branches,
482) -> Result<HashMap<String, BranchExclusivity>, DagError> {
483    let mut exclusive_to: HashMap<String, BranchExclusivity> = HashMap::new();
484
485    // Validate every branch target actually names a real node before
486    // seeding anything — an undeclared target is a build-time error (the
487    // node it should have gated instead runs unconditionally, which is
488    // exactly the silent-wrong-behavior this typechecker exists to prevent),
489    // named against the decision (branching node) that references it.
490    for (decision, labels) in &branches.decisions {
491        for (label, target) in labels {
492            if graph.get(target).is_none() {
493                return Err(DagError::UnknownReference {
494                    node: decision.clone(),
495                    referenced: target.clone(),
496                });
497            }
498            exclusive_to.insert(
499                target.clone(),
500                BranchExclusivity {
501                    decision: decision.clone(),
502                    label: label.clone(),
503                },
504            );
505        }
506    }
507
508    // Propagate: repeat until no new node gains an exclusivity marker.
509    // O(n^2) worst case, fine at this scale (specs have tens of nodes, not
510    // thousands) — larger-scale fan-out is explicitly deferred to a future
511    // cycle, not this one.
512    loop {
513        let mut changed = false;
514        for (name, node) in &graph.nodes {
515            let Some(expr) = &node.input else { continue };
516            let mut found: Option<BranchExclusivity> = None;
517            for referenced in referenced_nodes(expr) {
518                if let Some(ex) = exclusive_to.get(referenced) {
519                    match &found {
520                        None => found = Some(ex.clone()),
521                        Some(existing)
522                            if existing.decision == ex.decision && existing.label != ex.label =>
523                        {
524                            return Err(DagError::ConflictingBranchFanIn {
525                                node: name.clone(),
526                                decision: ex.decision.clone(),
527                                label_a: existing.label.clone(),
528                                label_b: ex.label.clone(),
529                            });
530                        }
531                        _ => {}
532                    }
533                }
534            }
535            if let Some(ex) = found {
536                if exclusive_to.insert(name.clone(), ex).is_none() {
537                    changed = true;
538                }
539            }
540        }
541        if !changed {
542            break;
543        }
544    }
545
546    Ok(exclusive_to)
547}
