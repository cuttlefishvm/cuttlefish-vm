1//! The reactor loop: the host drives the guest, one command at a time.
2//!
3//! This module is the shape of the whole system. The guest never calls the host
4//! and waits — it returns a [`Command`], the host carries it out, and the host
5//! steps the guest again with the resulting [`Event`]. Two things follow from
6//! that inversion, and both are why it is worth the awkwardness:
7//!
8//! - **Cancellation needs no guest cooperation.** The host simply stops
9//!   stepping. A guest cannot ignore, delay, or trap its way out of it.
10//! - **Every iteration is observable.** Progress, token counts, and capability
11//!   decisions all pass through the host, even for a block whose internal loop
12//!   the DAG cannot see.
13//!
14//! The alternative — host functions the guest imports and blocks on — is not
15//! merely less tidy, it does not work: a single-threaded core-wasm guest offers
16//! no execution context for the host to call back into, and the wasmtime `Store`
17//! is `!Sync` while inference must run on a separate thread.
18

45/// Everything needed to run one job.
46pub struct JobSpec {
47    /// The checked graph, in topological order — safe to execute
48    /// front-to-back, threading `outputs` forward.
49    pub nodes: Vec<crate::dag::CheckedNode>,
50    /// Which nodes are exclusive to which branch decision+label — see
51    /// `crate::dag::BranchExclusivity`.
52    pub exclusive_to: std::collections::HashMap<String, crate::dag::BranchExclusivity>,
53    /// The job's input, handed to every entry node (a node with no `input`
54    /// expression — `node.input.is_none()`).
55    pub input: serde_json::Value,
56    /// What this job is permitted to reach.
57    pub caps: Capabilities,

60/// Pointer width of a guest module, read from the module rather than assumed.
61///
62/// Only [`Abi::W32`] is supported today; 64-bit guests are rejected with a clear
63/// message. The enum exists anyway so that adding wasm64 later is a new arm plus
64/// a second set of [`TypedFunc`] signatures, rather than a hunt through this
65/// file for every place a pointer was assumed to be four bytes wide.
66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67pub enum Abi {
68    /// 32-bit linear memory.
69    W32,
70    /// 64-bit linear memory (memory64).
71    W64,
72}
73
74impl Abi {
75    /// Size of one pointer-sized field, and so half a descriptor.
76    fn ptr_size(self) -> usize {
77        match self {
78            Abi::W32 => 4,
79            Abi::W64 => 8,
80        }
81    }
82}

84struct Guest {
85    store: Store<()>,
86    memory: Memory,
87    abi: Abi,
88    alloc: TypedFunc<u32, u32>,
89    init: TypedFunc<(u32, u32), u32>,
90    step: TypedFunc<(u32, u32), u32>,
91    on_token: Option<TypedFunc<(u32, u32), i32>>,
92}
93
94impl Guest {
95    fn new(
96        engine: &Engine,
97        cache: &crate::module_cache::ModuleCache,
98        module_bytes: &[u8],
99    ) -> anyhow::Result<Self> {
100        let module = cache.compile(engine, module_bytes)?;
101
102        // An empty linker, deliberately. Guest blocks are built for
103        // `wasm32-unknown-unknown` and import nothing at all — a wasip1 guest
104        // would drag in `fd_write` and `proc_exit` through its panic path alone
105        // and fail to instantiate here.
106        let linker: Linker<()> = Linker::new(engine);
107        let mut store = Store::new(engine, ());
108        let instance: Instance = linker.instantiate(&mut store, &module)?;
109
110        let memory = instance
111            .get_memory(&mut store, "memory")
112            .ok_or_else(|| anyhow::anyhow!("guest exports no memory"))?;
113
114        // Width comes from the module itself. A 64-bit guest exports `cf_init`
115        // as `(i64, i64) -> i64`, so the typed lookups below would otherwise
116        // fail with a signature mismatch that says nothing about the real cause.
117        let abi = if memory.ty(&store).is_64() {
118            Abi::W64
119        } else {
120            Abi::W32
121        };
122        if abi == Abi::W64 {
123            anyhow::bail!("guest uses 64-bit memory; only 32-bit guests are supported");
124        }
125
126        Ok(Self {
127            alloc: instance.get_typed_func(&mut store, "cf_alloc")?,
128            init: instance.get_typed_func(&mut store, "cf_init")?,
129            step: instance.get_typed_func(&mut store, "cf_step")?,
130            // Optional: a block indifferent to streaming need not export it.
131            on_token: instance.get_typed_func(&mut store, "cf_on_token").ok(),
132            memory,
133            abi,
134            store,
135        })
136    }
137
138    fn write(&mut self, bytes: &[u8]) -> anyhow::Result<(u32, u32)> {
139        let len = bytes.len() as u32;
140        let ptr = self.alloc.call(&mut self.store, len)?;
141        self.memory.write(&mut self.store, ptr as usize, bytes)?;
142        Ok((ptr, len))
143    }
144
145    /// Read the descriptor the guest returned, then the payload it points at.
146    ///
147    /// Two reads rather than unpacking one integer — the cost of keeping these
148    /// signatures identical across pointer widths.
149    fn read_desc(&mut self, desc_ptr: u32) -> anyhow::Result<Vec<u8>> {
150        let w = self.abi.ptr_size();
151        let mut desc = vec![0u8; 2 * w];
152        self.memory
153            .read(&mut self.store, desc_ptr as usize, &mut desc)?;
154
155        let field = |bytes: &[u8]| -> u64 {
156            match w {
157                4 => u32::from_le_bytes(bytes.try_into().expect("4 bytes")) as u64,
158                _ => u64::from_le_bytes(bytes.try_into().expect("8 bytes")),
159            }
160        };
161        let ptr = field(&desc[..w]) as usize;
162        let len = field(&desc[w..]) as usize;
163
164        let mut buf = vec![0u8; len];
165        self.memory.read(&mut self.store, ptr, &mut buf)?;
166        Ok(buf)
167    }
168
169    fn call_init(&mut self, input: &serde_json::Value) -> anyhow::Result<Command> {
170        let bytes = serde_json::to_vec(input)?;
171        let (ptr, len) = self.write(&bytes)?;
172        let desc = self.init.call(&mut self.store, (ptr, len))?;
173        Ok(serde_json::from_slice(&self.read_desc(desc)?)?)
174    }
175
176    fn call_step(&mut self, event: &Event) -> anyhow::Result<Command> {
177        let bytes = serde_json::to_vec(event)?;
178        let (ptr, len) = self.write(&bytes)?;
179        let desc = self.step.call(&mut self.store, (ptr, len))?;
180        Ok(serde_json::from_slice(&self.read_desc(desc)?)?)
181    }
182
183    /// Ask the guest whether generation should continue.
184    fn call_on_token(&mut self, token: &str) -> anyhow::Result<bool> {
185        // Cloned rather than moved: wasmtime's TypedFunc is Clone but not Copy,
186        // and cloning also ends the borrow of `self` before `write` needs it
187        // mutably.
188        let Some(f) = self.on_token.clone() else {
189            return Ok(true);
190        };
191        let (ptr, len) = self.write(token.as_bytes())?;
192        Ok(f.call(&mut self.store, (ptr, len))? == 0)
193    }
194}

196/// Read a block's declared signature out of a compiled module.
197///
198/// Instantiates the module and calls its `cf_signature` export. That is heavier
199/// than parsing a sidecar file, and it is the point: the answer comes from the
200/// artifact that will actually run, so it cannot describe a different version of
201/// the block than the one being checked.
202///
203/// A block built before signatures existed has no such export. That is not an
204/// error — it reports the permissive default, so an older block still composes,
205/// just without the seam being checked.
206pub fn read_signature(
207    engine: &Engine,
208    module_bytes: &[u8],
209) -> anyhow::Result<cuttlefish_abi::Signature> {
210    let permissive = cuttlefish_abi::Signature {
211        input: cuttlefish_abi::Ty::Json,
212        output: cuttlefish_abi::Ty::Json,
213    };
214
215    // Deliberately does not go through `Guest`, which requires the whole reactor
216    // — alloc, init, step. Reading a declaration should not demand that a module
217    // be runnable: a block missing an export has a real problem, but it is one
218    // worth reporting when the job runs, with the job's error handling, rather
219    // than as a confusing failure during a typecheck.
220    let module = Module::new(engine, module_bytes)?;
221    let linker: Linker<()> = Linker::new(engine);
222    let mut store = Store::new(engine, ());
223    let instance = linker.instantiate(&mut store, &module)?;
224
225    let Ok(signature) = instance.get_typed_func::<(), u32>(&mut store, "cf_signature") else {
226        return Ok(permissive);
227    };
228    let Some(memory) = instance.get_memory(&mut store, "memory") else {
229        return Ok(permissive);
230    };
231
232    let desc_ptr = signature.call(&mut store, ())? as usize;
233    let mut desc = [0u8; 8];
234    memory.read(&mut store, desc_ptr, &mut desc)?;
235    let ptr = u32::from_le_bytes(desc[..4].try_into().expect("4 bytes")) as usize;
236    let len = u32::from_le_bytes(desc[4..].try_into().expect("4 bytes")) as usize;
237
238    let mut buf = vec![0u8; len];
239    memory.read(&mut store, ptr, &mut buf)?;
240    Ok(serde_json::from_slice(&buf)?)
241}

255/// Whether an `InputExpr` (transitively) references any node in `skipped`.
256fn references_any(
257    expr: &cuttlefish_core::graph::InputExpr,
258    skipped: &std::collections::HashSet<String>,
259) -> bool {
260    use cuttlefish_core::graph::InputExpr;
261    match expr {
262        InputExpr::FromNode(n) => skipped.contains(n),
263        InputExpr::Record(fields) => fields.values().any(|e| references_any(e, skipped)),
264        InputExpr::List(items) => items.iter().any(|e| references_any(e, skipped)),
265    }
266}

268/// Compose an `InputExpr` into an actual JSON value, by looking up each
269/// referenced node's already-produced output. Mirrors `dag::evaluate_expr_ty`
270/// (which does the same composition at the *type* level, at check time) —
271/// this is the runtime analogue.
272fn evaluate_input(
273    expr: &cuttlefish_core::graph::InputExpr,
274    outputs: &std::collections::HashMap<String, serde_json::Value>,
275) -> serde_json::Value {
276    use cuttlefish_core::graph::InputExpr;
277    match expr {
278        InputExpr::FromNode(n) => outputs.get(n).cloned().unwrap_or(serde_json::Value::Null),
279        InputExpr::Record(fields) => {
280            let mut map = serde_json::Map::new();
281            for (k, v) in fields {
282                map.insert(k.clone(), evaluate_input(v, outputs));
283            }
284            serde_json::Value::Object(map)
285        }
286        InputExpr::List(items) => {
287            serde_json::Value::Array(items.iter().map(|e| evaluate_input(e, outputs)).collect())
288        }
289    }
290}

304/// Drive one job to completion.
305///
306/// Always returns an [`Envelope`]; failures are values, not errors, because the
307/// caller has to report *something* to whoever submitted the job.
308///
309/// `ledger` is consulted before any resume-sensitive decision (branch-skip,
310/// transitive-skip, or actually running a node) and written to immediately
311/// after that decision is made, so that a process restart mid-job — a later
312/// task's concern, not this function's — can resume from exactly the state
313/// this function left behind. The whole body runs inside a single labeled
314/// block (`'run: { ... }`) so that every exit path, however it got there,
315/// still reaches the one `ledger.finish(...)` call at the end.
316pub async fn run_job(
317    engine: Arc<Engine>,
318    backend: Arc<dyn InferBackend>,
319    job: JobSpec,
320    events: mpsc::Sender<JobEvent>,
321    cancel: CancellationToken,
322    ledger: &crate::ledger::Ledger,
323    cache: &crate::module_cache::ModuleCache,
324) -> Envelope {
325    let started = Instant::now();
326    let mut usage = Usage {
327        model: backend.model_name(),
328        ..Usage::default()
329    };
330
331    // Dropped when this function returns, closing every file the job opened.
332    // That job-scoped lifetime is what makes handles unforgeable across jobs.
333    //
334    // Shared across stages on purpose: a handle produced by one block — a
335    // rendered page, say — stays usable by the next. Confining it to one stage
336    // would make a pipeline strictly weaker than a single block that did the
337    // same work, while adding nothing, since the job boundary is what the
338    // security property rests on.
339    let mut handles = Handles::default();
340
341    let envelope = 'run: {
342        if job.nodes.is_empty() {
343            usage.duration_ms = started.elapsed().as_millis() as u64;
344            break 'run fail(
345                error_codes::SCHEMA_VALIDATION_FAILED,
346                "this job has no nodes to run",
347                usage,
348            );
349        }
350
351        let total = job.nodes.len();
352        let mut outputs: std::collections::HashMap<String, serde_json::Value> =
353            std::collections::HashMap::new();
354        let mut skipped: std::collections::HashSet<String> = std::collections::HashSet::new();
355        let mut route_taken: std::collections::HashMap<String, String> =
356            std::collections::HashMap::new();
357
358        for (index, node) in job.nodes.iter().enumerate() {
359            // Resume: a node the ledger already marked skipped stays skipped
360            // — its branch decision is not re-evaluated.
361            match ledger.is_skipped(&node.name) {
362                Ok(true) => {
363                    skipped.insert(node.name.clone());
364                    continue;
365                }
366                Ok(false) => {}
367                Err(e) => {
368                    usage.duration_ms = started.elapsed().as_millis() as u64;
369                    break 'run fail(
370                        error_codes::SCHEMA_VALIDATION_FAILED,
371                        format!("reading ledger skip state for node `{}`: {e}", node.name),
372                        usage,
373                    );
374                }
375            }
376
377            // Resume: a completed checkpoint means reuse the cached output
378            // instead of re-running.
379            match ledger.get_completed(&node.name) {
380                Ok(Some(cached)) => {
381                    // If this was a branches decision node, its route must be
382                    // recorded in `route_taken` here too — not just after a
383                    // fresh `run_stage` call below — or a downstream,
384                    // not-yet-reached branch-exclusive node would see no
385                    // recorded decision at all on a resumed run and
386                    // (incorrectly) never be skipped. The checkpoint only
387                    // ever holds a value that already passed this same route
388                    // validation the first time it ran, so this is purely
389                    // re-deriving `route_taken`, never re-validating.
390                    if let Some(route) = cached.get("route").and_then(|v| v.as_str()) {
391                        route_taken.insert(node.name.clone(), route.to_string());
392                    }
393                    outputs.insert(node.name.clone(), cached);
394                    continue;
395                }
396                Ok(None) => {}
397                Err(e) => {
398                    usage.duration_ms = started.elapsed().as_millis() as u64;
399                    break 'run fail(
400                        error_codes::SCHEMA_VALIDATION_FAILED,
401                        format!("reading ledger checkpoint for node `{}`: {e}", node.name),
402                        usage,
403                    );
404                }
405            }
406
407            // Tell watchers which node is running. A pipeline that stalls is
408            // much easier to diagnose when the stream says where.
409            if total > 1 {
410                let _ = events
411                    .send(JobEvent::Progress(serde_json::json!({
412                        "stage": index + 1,
413                        "of": total,
414                        "node": node.name,
415                    })))
416                    .await;
417            }
418
419            // Fresh branch-skip decision (only reached if the ledger had no
420            // recorded state for this node — i.e. this is either a fresh
421            // run, or a resumed run that hadn't reached this node yet).
422            //
423            // Branch-skip: this node is exclusive to a decision+label, and
424            // that decision's chosen route (already recorded earlier in this
425            // same loop, since the branching node is always topologically
426            // before the nodes exclusive to its labels) doesn't match.
427            if let Some(ex) = job.exclusive_to.get(&node.name) {
428                if let Some(taken) = route_taken.get(&ex.decision) {
429                    if taken != &ex.label {
430                        skipped.insert(node.name.clone());
431                        if let Err(e) = ledger.write_skipped(&node.name) {
432                            usage.duration_ms = started.elapsed().as_millis() as u64;
433                            break 'run fail(
434                                error_codes::SCHEMA_VALIDATION_FAILED,
435                                format!("recording skip for node `{}` in ledger: {e}", node.name),
436                                usage,
437                            );
438                        }
439                        continue;
440                    }
441                }
442            }
443            // Transitive skip: this node's input needs a skipped node's
444            // output.
445            if let Some(expr) = &node.input {
446                if references_any(expr, &skipped) {
447                    skipped.insert(node.name.clone());
448                    if let Err(e) = ledger.write_skipped(&node.name) {
449                        usage.duration_ms = started.elapsed().as_millis() as u64;
450                        break 'run fail(
451                            error_codes::SCHEMA_VALIDATION_FAILED,
452                            format!("recording skip for node `{}` in ledger: {e}", node.name),
453                            usage,
454                        );
455                    }
456                    continue;
457                }
458            }
459
460            let node_input = match &node.input {
461                None => job.input.clone(),
462                Some(expr) => evaluate_input(expr, &outputs),
463            };
464
465            // Fan-out: this node runs once per manifest line rather than
466            // once, with each item checkpointed independently. Handled by
467            // its own helper because it is a genuinely different execution
468            // shape, not a variation on the repeat_until loop below (which
469            // it is parse-time forbidden to combine with).
470            if node.over.is_some() {
471                let collected = match run_fanout_node(
472                    &engine,
473                    cache,
474                    &backend,
475                    node,
476                    &job.caps,
477                    &mut handles,
478                    &events,
479                    &cancel,
480                    &mut usage,
481                    started,
482                    index,
483                    total,
484                    ledger,
485                )
486                .await
487                {
488                    Ok(v) => v,
489                    Err(envelope) => break 'run envelope,
490                };
491                if let Err(e) = ledger.write_completed(&node.name, &collected) {
492                    usage.duration_ms = started.elapsed().as_millis() as u64;
493                    break 'run fail(
494                        error_codes::SCHEMA_VALIDATION_FAILED,
495                        format!(
496                            "recording checkpoint for node `{}` in ledger: {e}",
497                            node.name
498                        ),
499                        usage,
500                    );
501                }
502                outputs.insert(node.name.clone(), collected);
503                continue;
504            }
505
506            // Bounded loop: repeat_until re-invokes this node on its own
507            // prior output, up to max_iterations. A node without
508            // repeat_until runs exactly once (the loop's `None` arm breaks
509            // immediately).
510            let mut current_input = node_input;
511            let mut iterations: u32 = 0;
512            let result = loop {
513                let r = match run_stage(
514                    &engine,
515                    cache,
516                    &backend,
517                    &node.module_bytes,
518                    current_input.clone(),
519                    node.script.as_deref(),
520                    &job.caps,
521                    &mut handles,
522                    &events,
523                    &cancel,
524                    &mut usage,
525                    started,
526                    index,
527                )
528                .await
529                {
530                    Ok(v) => v,
531                    Err(envelope) => break 'run envelope,
532                };
533                match &node.repeat_until {
534                    None => break r,
535                    Some(field) => {
536                        iterations += 1;
537                        let done = r.get(field).and_then(|v| v.as_str()) == Some("done");
538                        if done {
539                            break r;
540                        }
541                        let max = node
542                            .max_iterations
543                            .expect("repeat_until requires max_iterations, enforced at parse time");
544                        if iterations >= max {
545                            usage.duration_ms = started.elapsed().as_millis() as u64;
546                            break 'run fail(
547                                error_codes::SCHEMA_VALIDATION_FAILED,
548                                format!(
549                                    "node `{}` did not reach repeat_until=\"done\" within max_iterations={max}",
550                                    node.name
551                                ),
552                                usage,
553                            );
554                        }
555                        current_input = r;
556                    }
557                }
558            };
559
560            // Nothing checked a block's *actual* output against what it
561            // declared until this existed: a Script or Rust block could
562            // claim `{summary: text}` and return `{text: "..."}` and the
563            // job would still complete "successfully", silently handing
564            // whatever consumed `summary` a `null`. `matches_value` is
565            // deliberately permissive about `bytes`/`image`/`document` (see
566            // its own doc comment) so this only ever catches genuine shape
567            // mismatches, not false positives on types this protocol has no
568            // fixed JSON encoding for.
569            if !node.signature.output.matches_value(&result) {
570                usage.duration_ms = started.elapsed().as_millis() as u64;
571                break 'run fail(
572                    error_codes::SCHEMA_VALIDATION_FAILED,
573                    format!(
574                        "node `{}` declared its output as `{}` but produced {result}, which \
575                         doesn't match",
576                        node.name, node.signature.output
577                    ),
578                    usage,
579                );
580            }
581
582            // A branching node: read its `route` field and record the
583            // decision for later nodes' branch-skip check (top of this
584            // loop).
585            //
586            // Whether this node is actually a `branches` decision at all is
587            // determined from `job.exclusive_to` rather than by threading
588            // `Branches` itself into `JobSpec`: `dag::compute_branch_exclusivity`
589            // seeds an entry for every `(label, target)` pair of every declared
590            // decision, so the set of `label`s across all `exclusive_to` entries
591            // whose `decision` names this node IS exactly this decision's full,
592            // valid label set. An unmatched or missing `route` on a genuine
593            // decision node is a job failure — loud, not a silent no-op — per
594            // the design spec's "Conditional dispatch" section.
595            let valid_labels: Vec<&str> = job
596                .exclusive_to
597                .values()
598                .filter(|ex| ex.decision == node.name)
599                .map(|ex| ex.label.as_str())
600                .collect();
601            if !valid_labels.is_empty() {
602                match result.get("route").and_then(|v| v.as_str()) {
603                    Some(route) if valid_labels.contains(&route) => {
604                        route_taken.insert(node.name.clone(), route.to_string());
605                    }
606                    Some(route) => {
607                        usage.duration_ms = started.elapsed().as_millis() as u64;
608                        break 'run fail(
609                            error_codes::SCHEMA_VALIDATION_FAILED,
610                            format!(
611                                "node `{}` produced route \"{route}\", which doesn't match any \
612                                 declared label for this branches decision",
613                                node.name
614                            ),
615                            usage,
616                        );
617                    }
618                    None => {
619                        usage.duration_ms = started.elapsed().as_millis() as u64;
620                        break 'run fail(
621                            error_codes::SCHEMA_VALIDATION_FAILED,
622                            format!(
623                                "node `{}` is a branches decision but its output has no `route` field",
624                                node.name
625                            ),
626                            usage,
627                        );
628                    }
629                }
630            } else if let Some(route) = result.get("route").and_then(|v| v.as_str()) {
631                // Not a declared decision node, but its output happens to
632                // carry a route field anyway — harmless, record it
633                // defensively (no downstream node's exclusive_to can
634                // reference this decision name if it isn't a real decision,
635                // so this is inert either way, just avoids silently
636                // dropping information that happens to be present).
637                route_taken.insert(node.name.clone(), route.to_string());
638            }
639
640            // Checkpoint a genuinely-executed node's result before recording
641            // it in `outputs` — the ledger write must happen before the
642            // in-memory outputs map update so a crash between the two can't
643            // leave the ledger silently behind the in-memory state (the
644            // in-memory state doesn't survive a crash anyway, so
645            // ledger-first is the only order that matters for durability;
646            // the reverse order would just be a smaller window for the same
647            // underlying property, not correctness).
648            if let Err(e) = ledger.write_completed(&node.name, &result) {
649                usage.duration_ms = started.elapsed().as_millis() as u64;
650                break 'run fail(
651                    error_codes::SCHEMA_VALIDATION_FAILED,
652                    format!(
653                        "recording checkpoint for node `{}` in ledger: {e}",
654                        node.name
655                    ),
656                    usage,
657                );
658            }
659            outputs.insert(node.name.clone(), result);
660        }
661
662        usage.duration_ms = started.elapsed().as_millis() as u64;
663
664        // What is "the" job result for a graph? Convention: the output of the
665        // LAST node in topological order that wasn't skipped. This exactly
666        // matches today's linear-pipeline behavior in the degenerate case (a
667        // linear chain's last node in topo order is its sole sink), and
668        // generalizes sensibly to a branching graph (exactly one path executes,
669        // so there's still a well-defined "last node that actually ran") and to
670        // a fan-in graph without branches (the join/sink node is last in topo
671        // order, since nothing depends on it).
672        //
673        // Known limitation: for a graph with two or more independent sinks (no
674        // edge between them at all — `cuttlefishd`'s daemon path genuinely
675        // accepts such graphs, unlike `cuttlefish build`, which restricts itself
676        // to linear graphs), "last in topological order" is decided by the
677        // topological sort's tie-break (alphabetical node name among ready
678        // nodes), not by any deliberate semantic answer about which sink's
679        // output should represent the job. Don't mistake that tie-break for a
680        // considered multi-sink policy — it isn't one.
681        let result = job
682            .nodes
683            .iter()
684            .rev()
685            .find_map(|n| outputs.get(&n.name).cloned());
686
687        match result {
688            Some(value) => Envelope {
689                status: JobStatus::Completed,
690                result: Some(value),
691                error: None,
692                usage,
693            },
694            None => fail(
695                error_codes::SCHEMA_VALIDATION_FAILED,
696                "every node in this job was skipped; there is no result",
697                usage,
698            ),
699        }
700    };
701
702    let ledger_status = match envelope.status {
703        JobStatus::Completed => "completed",
704        JobStatus::Failed => "failed",
705        JobStatus::Cancelled => "cancelled",
706        _ => "running", // shouldn't occur — defensive only
707    };
708    if let Err(e) = ledger.finish(ledger_status) {
709        // A ledger write failure at this final step doesn't invalidate the
710        // job's already-computed, correct result — losing it here means a
711        // wrong Interrupted-detection on next restart, not a wrong result
712        // now. Loud enough to notice, not loud enough to discard real work
713        // that already succeeded or failed for its own, unrelated reason.
714        eprintln!("warning: failed to record job {ledger_status} status in ledger: {e}");
715    }
716    envelope
717}
718
719/// Run one block to completion, returning what it produced.
720///
721/// `Err` carries a finished [`Envelope`]: a stage that fails ends the whole job,
722/// because a later stage's input is the earlier one's output and there is
723/// nothing sensible to feed it.
724/// Run one fan-out node: its block once per manifest line, each item
725/// checkpointed independently, then the results materialized for whatever
726/// consumes them downstream.
727///
728/// Returns the collection record described by
729/// [`crate::dag::fanout_collection_ty`] — deliberately not any one item's
730/// result, since downstream consumes all of them.
731///
732/// # Why the ledger is authoritative and the files are a projection
733///
734/// Items are recorded in SQLite as they conclude, and `results.jsonl` /
735/// `failures.jsonl` are written once at the end by reading those rows back.
736/// Appending to the text files as items finished would be simpler, but a
737/// crash mid-append leaves a torn final line that resume then has to
738/// reconcile against the ledger. Projecting at the end makes that class of
739/// bug unrepresentable: SQLite is already transactional, so let it be the
740/// thing that's true.
741#[allow(clippy::too_many_arguments)]
742async fn run_fanout_node(
743    engine: &Engine,
744    cache: &crate::module_cache::ModuleCache,
745    backend: &Arc<dyn InferBackend>,
746    node: &crate::dag::CheckedNode,
747    caps: &Capabilities,
748    handles: &mut Handles,
749    events: &mpsc::Sender<JobEvent>,
750    cancel: &CancellationToken,
751    usage: &mut Usage,
752    started: Instant,
753    index: usize,
754    total: usize,
755    ledger: &crate::ledger::Ledger,
756) -> Result<serde_json::Value, Envelope> {
757    use sha2::{Digest, Sha256};
758
759    let manifest_path = node
760        .over
761        .as_ref()
762        .expect("run_fanout_node is only called for a node with `over`");
763    let node_name = node.name.as_str();
764    // NOT `node.signature.output` — that has already been replaced with the
765    // collection record for downstream typing, so validating an item against
766    // it would reject every single item.
767    let item_output = node
768        .item_output
769        .as_ref()
770        .unwrap_or(&node.signature.output)
771        .clone();
772
773    let bail = |message: String, usage: &mut Usage| -> Envelope {
774        usage.duration_ms = started.elapsed().as_millis() as u64;
775        fail(
776            error_codes::SCHEMA_VALIDATION_FAILED,
777            message,
778            usage.clone(),
779        )
780    };
781
782    // --- Read and validate the manifest up front -------------------------
783    //
784    // A malformed manifest is an authoring error, so it fails the whole job
785    // before any item runs, rather than surfacing as N mysterious item
786    // failures partway through.
787    let bytes = std::fs::read(manifest_path).map_err(|e| {
788        bail(
789            format!(
790                "node `{node_name}`: reading fan-out manifest {}: {e}",
791                manifest_path.display()
792            ),
793            usage,
794        )
795    })?;
796
797    let text = String::from_utf8(bytes.clone()).map_err(|e| {
798        bail(
799            format!(
800                "node `{node_name}`: fan-out manifest {} is not valid UTF-8: {e}",
801                manifest_path.display()
802            ),
803            usage,
804        )
805    })?;
806
807    let mut items: Vec<serde_json::Value> = Vec::new();
808    for (i, line) in text.lines().enumerate() {
809        if line.trim().is_empty() {
810            continue;
811        }
812        match serde_json::from_str(line) {
813            Ok(v) => items.push(v),
814            Err(e) => {
815                return Err(bail(
816                    format!(
817                        "node `{node_name}`: fan-out manifest {} line {} is not valid JSON: {e}",
818                        manifest_path.display(),
819                        i + 1
820                    ),
821                    usage,
822                ))
823            }
824        }
825    }
826
827    if items.is_empty() {
828        return Err(bail(
829            format!(
830                "node `{node_name}`: fan-out manifest {} is empty — zero items almost always \
831                 means the step that produced it failed, and reducing over nothing would \
832                 silently look like success",
833                manifest_path.display()
834            ),
835            usage,
836        ));
837    }
838
839    // --- Pin this node to the manifest it ran against --------------------
840    //
841    // An item_index is only meaningful relative to one manifest, and no
842    // graph fingerprint can see an edit to the manifest *file*.
843    let digest = format!("{:x}", Sha256::digest(&bytes));
844    match ledger.check_or_record_manifest(node_name, &digest, items.len()) {
845        Ok(Ok(())) => {}
846        Ok(Err(previous)) => {
847            return Err(bail(
848                format!(
849                    "node `{node_name}`: fan-out manifest {} has changed since this job first \
850                     ran (was {previous}, now {digest}) — recorded item indices no longer refer \
851                     to the same inputs, so resuming would pair results with the wrong items; \
852                     re-submit the job instead",
853                    manifest_path.display()
854                ),
855                usage,
856            ))
857        }
858        Err(e) => {
859            return Err(bail(
860                format!("node `{node_name}`: recording fan-out manifest digest: {e}"),
861                usage,
862            ))
863        }
864    }
865
866    // --- Run each item ---------------------------------------------------
867    let mut succeeded = 0usize;
868    let mut failed = 0usize;
869
870    for (item_index, item_input) in items.iter().enumerate() {
871        // Cancellation is checked here, not only inside run_stage: without
872        // it a cancelled 500-item campaign would keep starting new items,
873        // finishing only once the manifest ran out.
874        if cancel.is_cancelled() {
875            usage.duration_ms = started.elapsed().as_millis() as u64;
876            return Err(Envelope {
877                status: JobStatus::Cancelled,
878                result: None,
879                error: None,
880                usage: usage.clone(),
881            });
882        }
883
884        // Resume: an item that already concluded — either way — is not run
885        // again. An item that was merely in flight when a previous run died
886        // left no row, so it lands here and runs now.
887        match ledger.item_concluded(node_name, item_index) {
888            Ok(true) => {
889                if ledger
890                    .get_item_completed(node_name, item_index)
891                    .unwrap_or(None)
892                    .is_some()
893                {
894                    succeeded += 1;
895                } else {
896                    failed += 1;
897                }
898                continue;
899            }
900            Ok(false) => {}
901            Err(e) => {
902                return Err(bail(
903                    format!("node `{node_name}`: reading item {item_index} from ledger: {e}"),
904                    usage,
905                ))
906            }
907        }
908
909        let _ = events
910            .send(JobEvent::Progress(serde_json::json!({
911                "stage": index + 1,
912                "of": total,
913                "node": node_name,
914                "item": item_index,
915                "items": items.len(),
916                "succeeded": succeeded,
917                "failed": failed,
918            })))
919            .await;
920
921        // Each item's input must satisfy what the block declared. A single
922        // mismatched line is a data-quality problem, not an authoring one,
923        // so it fails that item and the run continues.
924        if !node.signature.input.matches_value(item_input) {
925            failed += 1;
926            let message = format!(
927                "item {item_index} does not match the block's declared input `{}`",
928                node.signature.input
929            );
930            if let Err(e) = ledger.write_item_failed(node_name, item_index, &message) {
931                return Err(bail(
932                    format!("node `{node_name}`: recording item {item_index} failure: {e}"),
933                    usage,
934                ));
935            }
936            continue;
937        }
938
939        let outcome = run_stage(
940            engine,
941            cache,
942            backend,
943            &node.module_bytes,
944            item_input.clone(),
945            node.script.as_deref(),
946            caps,
947            handles,
948            events,
949            cancel,
950            usage,
951            started,
952            index,
953        )
954        .await;
955
956        match outcome {
957            Ok(value) if item_output.matches_value(&value) => {
958                succeeded += 1;
959                if let Err(e) = ledger.write_item_completed(node_name, item_index, &value) {
960                    return Err(bail(
961                        format!("node `{node_name}`: recording item {item_index} result: {e}"),
962                        usage,
963                    ));
964                }
965            }
966            Ok(value) => {
967                failed += 1;
968                let message = format!(
969                    "item {item_index} produced {value}, which doesn't match the block's \
970                     declared output `{item_output}`"
971                );
972                if let Err(e) = ledger.write_item_failed(node_name, item_index, &message) {
973                    return Err(bail(
974                        format!("node `{node_name}`: recording item {item_index} failure: {e}"),
975                        usage,
976                    ));
977                }
978            }
979            // A cancelled job must not be recorded as a per-item failure —
980            // nothing about the item was wrong, and doing so would make the
981            // cancellation permanent across a later resume.
982            Err(envelope) if envelope.status == JobStatus::Cancelled => return Err(envelope),
983            Err(envelope) => {
984                failed += 1;
985                let message = envelope
986                    .error
987                    .as_ref()
988                    .map(|e| format!("{}: {}", e.code, e.message))
989                    .unwrap_or_else(|| "item failed with no error detail".to_string());
990                if let Err(e) = ledger.write_item_failed(node_name, item_index, &message) {
991                    return Err(bail(
992                        format!("node `{node_name}`: recording item {item_index} failure: {e}"),
993                        usage,
994                    ));
995                }
996            }
997        }
998    }
999
1000    if succeeded == 0 {
1001        // Quote the first item's actual error. "All 500 items failed" with no
1002        // cause is a dead end, and when every item fails the same way — which
1003        // is the common case — the first one is the whole story.
1004        let first_error = ledger
1005            .concluded_items(node_name)
1006            .ok()
1007            .and_then(|items| items.into_iter().find_map(|(_, _, err)| err))
1008            .unwrap_or_else(|| "no error recorded".to_string());
1009        return Err(bail(
1010            format!(
1011                "node `{node_name}`: all {failed} fan-out item(s) failed — there is nothing for \
1012                 a downstream node to reduce over. First failure: {first_error}"
1013            ),
1014            usage,
1015        ));
1016    }
1017
1018    // --- Materialize the results from the ledger -------------------------
1019    //
1020    // Named per node: a graph may legally contain more than one fan-out
1021    // node, and flat results.jsonl / failures.jsonl would clobber.
1022    let results_dir = ledger.job_dir().join("results");
1023    std::fs::create_dir_all(&results_dir).map_err(|e| {
1024        bail(
1025            format!(
1026                "node `{node_name}`: creating {}: {e}",
1027                results_dir.display()
1028            ),
1029            usage,
1030        )
1031    })?;
1032    let results_path = results_dir.join(format!("{node_name}.results.jsonl"));
1033    let failures_path = results_dir.join(format!("{node_name}.failures.jsonl"));
1034
1035    let concluded = ledger.concluded_items(node_name).map_err(|e| {
1036        bail(
1037            format!("node `{node_name}`: reading concluded items from ledger: {e}"),
1038            usage,
1039        )
1040    })?;
1041
1042    let (mut results_out, mut failures_out) = (String::new(), String::new());
1043    for (item_index, output, error) in concluded {
1044        match (output, error) {
1045            (Some(value), _) => results_out.push_str(&format!(
1046                "{}\n",
1047                serde_json::json!({"item": item_index, "result": value})
1048            )),
1049            (None, Some(message)) => failures_out.push_str(&format!(
1050                "{}\n",
1051                serde_json::json!({"item": item_index, "error": message})
1052            )),
1053            (None, None) => {}
1054        }
1055    }
1056
1057    for (path, contents) in [
1058        (&results_path, &results_out),
1059        (&failures_path, &failures_out),
1060    ] {
1061        std::fs::write(path, contents).map_err(|e| {
1062            bail(
1063                format!("node `{node_name}`: writing {}: {e}", path.display()),
1064                usage,
1065            )
1066        })?;
1067    }
1068
1069    Ok(serde_json::json!({
1070        "results_path": results_path.to_string_lossy(),
1071        "failures_path": failures_path.to_string_lossy(),
1072        "succeeded": succeeded,
1073        "failed": failed,
1074    }))
1075}
1076
1077#[allow(clippy::too_many_arguments)]
1078async fn run_stage(
1079    engine: &Engine,
1080    cache: &crate::module_cache::ModuleCache,
1081    backend: &Arc<dyn InferBackend>,
1082    module_bytes: &[u8],
1083    input: serde_json::Value,
1084    script: Option<&str>,
1085    caps: &Capabilities,
1086    handles: &mut Handles,
1087    events: &mpsc::Sender<JobEvent>,
1088    cancel: &CancellationToken,
1089    usage: &mut Usage,
1090    started: Instant,
1091    stage_index: usize,
1092) -> Result<serde_json::Value, Envelope> {
1093    // Naming the stage turns "the job failed" into "the second block failed",
1094    // which is the difference between a usable error and a hunt.
1095    let blame = |message: String| -> String {
1096        if stage_index == 0 {
1097            message
1098        } else {
1099            format!("block {} of the pipeline: {message}", stage_index + 1)
1100        }
1101    };
1102    // Documents are read from their path rather than their descriptor — both
1103    // extraction and rendering want a file. Kept beside the handle table so the
1104    // two are dropped together at the end of the job.
1105    let mut doc_paths: std::collections::HashMap<u32, std::path::PathBuf> =
1106        std::collections::HashMap::new();
1107
1108    let mut guest = match Guest::new(engine, cache, module_bytes) {
1109        Ok(g) => g,
1110        Err(e) => {
1111            return Err(fail(
1112                error_codes::WASM_TRAP,
1113                blame(e.to_string()),
1114                usage.clone(),
1115            ))
1116        }
1117    };
1118
1119    // A Script-kind stage's `module_bytes` is always the shared interpreter
1120    // (see `pipeline::resolve_and_load`), which expects its script text
1121    // wrapped alongside the real job input — the interpreter itself never
1122    // receives the raw input directly, and this is the one place per job
1123    // where that wrapping actually happens, since the script is fixed at
1124    // catalog time but the input is only known per job.
1125    let input = match script {
1126        Some(script) => serde_json::json!({
1127            "__cuttlefish_script": script,
1128            "input": input,
1129        }),
1130        None => input,
1131    };
1132
1133    let mut command = match guest.call_init(&input) {
1134        Ok(c) => c,
1135        Err(e) => {
1136            return Err(fail(
1137                error_codes::WASM_TRAP,
1138                blame(e.to_string()),
1139                usage.clone(),
1140            ))
1141        }
1142    };
1143
1144    loop {
1145        if cancel.is_cancelled() {
1146            usage.duration_ms = started.elapsed().as_millis() as u64;
1147            return Err(cancelled(usage.clone(), "job cancelled"));
1148        }
1149
1150        let event = match command {
1151            // Ends this stage, not the job: the value becomes the next
1152            // block's input, or the job's result if this was the last.
1153            Command::Done { result } => return Ok(result),
1154            Command::Fail { code, message } => {
1155                usage.duration_ms = started.elapsed().as_millis() as u64;
1156                return Err(fail(&code, blame(message), usage.clone()));
1157            }
1158            Command::Emit { progress } => {
1159                let _ = events.send(JobEvent::Progress(progress)).await;
1160                Event::Emitted
1161            }
1162
1163            // The capability check lives here, at Open, and nowhere else. Slice
1164            // takes a handle rather than a path, and handles are job-scoped, so
1165            // there is no second place a path can enter the system.
1166            Command::Open { path } => {
1167                let p = std::path::PathBuf::from(&path);
1168                if !caps.allows_read(&p) {
1169                    usage.duration_ms = started.elapsed().as_millis() as u64;
1170                    return Err(fail(
1171                        error_codes::CAPABILITY_DENIED,
1172                        format!("read not permitted: {path}"),
1173                        usage.clone(),
1174                    ));
1175                }
1176                match handles.open(&p) {
1177                    Ok((handle, len, kind)) => {
1178                        // A PDF's page count and text layer need the whole file,
1179                        // which the handle layer deliberately does not read. Ask
1180                        // the document layer, and fall back to the plain kind if
1181                        // it cannot answer — a malformed PDF is still a file a
1182                        // block may want to read bytes from.
1183                        let kind = match kind {
1184                            cuttlefish_abi::MediaKind::Document { .. } => {
1185                                match crate::documents::inspect(&p) {
1186                                    Ok(info) => cuttlefish_abi::MediaKind::Document {
1187                                        pages: info.pages,
1188                                        has_text_layer: info.has_text_layer,
1189                                    },
1190                                    Err(_) => cuttlefish_abi::MediaKind::Binary,
1191                                }
1192                            }
1193                            other => other,
1194                        };
1195                        // Remember the path: rendering and text extraction work
1196                        // from a file, not from the open descriptor.
1197                        doc_paths.insert(handle, p.clone());
1198                        Event::Opened { handle, len, kind }
1199                    }
1200                    Err(e) => {
1201                        usage.duration_ms = started.elapsed().as_millis() as u64;
1202                        return Err(fail(
1203                            error_codes::CAPABILITY_DENIED,
1204                            e.to_string(),
1205                            usage.clone(),
1206                        ));
1207                    }
1208                }
1209            }
1210
1211            Command::Slice {
1212                handle,
1213                offset,
1214                len,
1215            } => match handles.slice(handle, offset, len) {
1216                Ok(w) => Event::Sliced {
1217                    text: w.text,
1218                    next_offset: w.next_offset,
1219                },
1220                Err(e) => {
1221                    usage.duration_ms = started.elapsed().as_millis() as u64;
1222                    return Err(fail(
1223                        error_codes::CAPABILITY_DENIED,
1224                        e.to_string(),
1225                        usage.clone(),
1226                    ));
1227                }
1228            },
1229
1230            Command::SliceBytes {
1231                handle,
1232                offset,
1233                len,
1234            } => match handles.slice_bytes(handle, offset, len) {
1235                Ok((bytes, next_offset)) => {
1236                    use base64::Engine;
1237                    Event::SlicedBytes {
1238                        bytes_base64: base64::engine::general_purpose::STANDARD.encode(&bytes),
1239                        next_offset,
1240                    }
1241                }
1242                Err(e) => {
1243                    usage.duration_ms = started.elapsed().as_millis() as u64;
1244                    return Err(fail(
1245                        error_codes::CAPABILITY_DENIED,
1246                        e.to_string(),
1247                        usage.clone(),
1248                    ));
1249                }
1250            },
1251
1252            Command::PageText { handle, page } => {
1253                let Some(path) = doc_paths.get(&handle).cloned() else {
1254                    usage.duration_ms = started.elapsed().as_millis() as u64;
1255                    return Err(fail(
1256                        error_codes::CAPABILITY_DENIED,
1257                        format!("no such handle: {handle}"),
1258                        usage.clone(),
1259                    ));
1260                };
1261                match crate::documents::page_text(&path, page) {
1262                    Ok(text) => Event::PageTexted { text },
1263                    Err(e) => {
1264                        usage.duration_ms = started.elapsed().as_millis() as u64;
1265                        return Err(fail(error_codes::UNSUPPORTED, e.to_string(), usage.clone()));
1266                    }
1267                }
1268            }
1269
1270            Command::PageImage { handle, page } => {
1271                let Some(path) = doc_paths.get(&handle).cloned() else {
1272                    usage.duration_ms = started.elapsed().as_millis() as u64;
1273                    return Err(fail(
1274                        error_codes::CAPABILITY_DENIED,
1275                        format!("no such handle: {handle}"),
1276                        usage.clone(),
1277                    ));
1278                };
1279                // A rendered page becomes a handle like any other, so it can be
1280                // named in Infer exactly as a file-backed image would be.
1281                match crate::documents::render_page(&path, page, RENDER_WIDTH) {
1282                    Ok(png) => {
1283                        let (handle, len) = handles.insert_bytes(
1284                            png,
1285                            cuttlefish_abi::MediaKind::Image {
1286                                format: "png".into(),
1287                            },
1288                        );
1289                        Event::PageImaged { handle, len }
1290                    }
1291                    Err(e) => {
1292                        usage.duration_ms = started.elapsed().as_millis() as u64;
1293                        return Err(fail(error_codes::UNSUPPORTED, e.to_string(), usage.clone()));
1294                    }
1295                }
1296            }
1297
1298            Command::Infer {
1299                prompt,
1300                max_tokens,
1301                images,
1302            } => {
1303                // Images are named by handle; the host loads the bytes, so they
1304                // never pass through guest memory.
1305                // Refuse rather than drop. Sending images to a backend that
1306                // cannot use them produces a confident answer about nothing,
1307                // which reads as a bad model rather than a misconfigured job —
1308                // and the caller has no way to tell the difference.
1309                if !images.is_empty() && !backend.supports_images() {
1310                    usage.duration_ms = started.elapsed().as_millis() as u64;
1311                    return Err(fail(
1312                        error_codes::UNSUPPORTED,
1313                        format!(
1314                            "this job supplied {} image(s), but the backend serving `{}` cannot \
1315                             accept them. Use a vision-capable model through the `ollama` \
1316                             provider, or change the block to send text only.",
1317                            images.len(),
1318                            backend.model_name()
1319                        ),
1320                        usage.clone(),
1321                    ));
1322                }
1323                let mut image_bytes = Vec::with_capacity(images.len());
1324                for handle in &images {
1325                    match handles.read_all(*handle) {
1326                        Ok(bytes) => image_bytes.push(bytes),
1327                        Err(e) => {
1328                            usage.duration_ms = started.elapsed().as_millis() as u64;
1329                            return Err(fail(
1330                                error_codes::CAPABILITY_DENIED,
1331                                e.to_string(),
1332                                usage.clone(),
1333                            ));
1334                        }
1335                    }
1336                }
1337
1338                // Tokens must reach the guest *while* generation runs, because
1339                // the guest's Stop verdict is what ends it early. The wasmtime
1340                // Store is !Sync and cannot be touched from inside the backend's
1341                // callback, so a channel carries tokens out and a shared flag
1342                // carries the verdict back — without sharing the Store.
1343                let (tx, mut rx) = mpsc::unbounded_channel::<String>();
1344                let stop = Arc::new(AtomicBool::new(false));
1345                let sink_stop = stop.clone();
1346                let mut sink = move |t: &str| {
1347                    tx.send(t.to_string()).is_ok() && !sink_stop.load(Ordering::Relaxed)
1348                };
1349
1350                let mut trap: Option<String> = None;
1351                let outcome: Option<anyhow::Result<InferResult>> = {
1352                    let request = InferRequest {
1353                        prompt: &prompt,
1354                        max_tokens,
1355                        images: &image_bytes,
1356                    };
1357                    let infer = backend.infer(request, &mut sink);
1358                    tokio::pin!(infer);
1359                    loop {
1360                        tokio::select! {
1361                            biased;
1362                            _ = cancel.cancelled() => break None,
1363                            Some(tok) = rx.recv() => {
1364                                let _ = events.send(JobEvent::Token(tok.clone())).await;
1365                                match guest.call_on_token(&tok) {
1366                                    Ok(true) => {}
1367                                    Ok(false) => stop.store(true, Ordering::Relaxed),
1368                                    Err(e) => {
1369                                        trap = Some(e.to_string());
1370                                        break None;
1371                                    }
1372                                }
1373                            }
1374                            r = &mut infer => break Some(r),
1375                        }
1376                    }
1377                };
1378
1379                if let Some(message) = trap {
1380                    usage.duration_ms = started.elapsed().as_millis() as u64;
1381                    return Err(fail(error_codes::WASM_TRAP, message, usage.clone()));
1382                }
1383
1384                // Tokens generated in the same poll as the last one are still
1385                // queued; forward them so the stream is complete.
1386                while let Ok(tok) = rx.try_recv() {
1387                    let _ = events.send(JobEvent::Token(tok)).await;
1388                }
1389
1390                match outcome {
1391                    None => {
1392                        usage.duration_ms = started.elapsed().as_millis() as u64;
1393                        return Err(cancelled(usage.clone(), "cancelled during inference"));
1394                    }
1395                    Some(Err(e)) => {
1396                        usage.duration_ms = started.elapsed().as_millis() as u64;
1397                        return Err(fail(
1398                            error_codes::MODEL_LOAD_FAILED,
1399                            e.to_string(),
1400                            usage.clone(),
1401                        ));
1402                    }
1403                    Some(Ok(r)) => {
1404                        usage.tokens_in += r.tokens_in;
1405                        usage.tokens_out += r.tokens_out;
1406                        Event::InferDone {
1407                            text: r.text,
1408                            tokens_out: r.tokens_out,
1409                        }
1410                    }
1411                }
1412            }
1413        };
1414
1415        command = match guest.call_step(&event) {
1416            Ok(c) => c,
1417            Err(e) => {
1418                usage.duration_ms = started.elapsed().as_millis() as u64;
1419                return Err(fail(
1420                    error_codes::WASM_TRAP,
1421                    blame(e.to_string()),
1422                    usage.clone(),
1423                ));
1424            }
1425        };
1426    }
1427}
