1//! Per-job durable checkpoint store. One SQLite file per job
2//! (`$CUTTLEFISH_HOME/jobs/<job_id>/ledger.sqlite`), matching the catalog's
3//! existing one-thing-per-file convention. See
4//! docs/superpowers/specs/2026-08-03-dag-core-design.md's "Durability model"
5//! for the full rationale — this module is purely storage; the resume
6//! decision logic (skip on completed/skipped, run everything else) lives in
7//! `crate::runner`.
8
9use rusqlite::Connection;
10use std::path::Path;
11use std::sync::Mutex;

13/// A job's own terminal status, as recorded in the ledger — distinct from
14/// per-node checkpoints, which alone can't tell "still running when the
15/// process died" apart from "finished cleanly."
16#[derive(Debug, Clone, Copy, PartialEq, Eq)]
17pub enum LedgerJobStatus {
18    /// The job has not yet called [`Ledger::finish`].
19    Running,
20    /// The job finished successfully.
21    Completed,
22    /// The job finished with an error.
23    Failed,
24    /// The job was cancelled before it finished.
25    Cancelled,
26}
27
28impl LedgerJobStatus {
29    fn from_str(s: &str) -> Self {
30        match s {
31            "completed" => Self::Completed,
32            "failed" => Self::Failed,
33            "cancelled" => Self::Cancelled,
34            _ => Self::Running,
35        }
36    }
37}

39/// A per-job checkpoint ledger, backed by a single SQLite file.
40///
41/// The connection is behind a `Mutex` purely to make `Ledger: Sync` —
42/// `rusqlite::Connection` is `Send` but not `Sync` (its statement cache uses
43/// unsynchronized interior mutability), and `run_job` holds a `&Ledger`
44/// across `.await` points in a task spawned onto a multi-threaded runtime,
45/// which requires the held reference to be `Send`, which in turn requires
46/// `Ledger: Sync`. There is normally only ever one writer (the job that owns
47/// this ledger), so contention is not a real concern; the lock exists to
48/// satisfy the type system's threading rules, not to arbitrate real
49/// concurrent access.
50pub struct Ledger {
51    conn: Mutex<Connection>,
52    job_dir: std::path::PathBuf,
53}
54
55/// Names the job this ledger belongs to without trying to render the SQLite
56/// connection, which is not `Debug`. Exists so callers can use
57/// `Result`-combinators like `expect_err` on [`Ledger::open`].
58impl std::fmt::Debug for Ledger {
59    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
60        f.debug_struct("Ledger")
61            .field("job_dir", &self.job_dir)
62            .finish_non_exhaustive()
63    }
64}

66/// Why a ledger could not be opened.
67#[derive(Debug, thiserror::Error)]
68pub enum LedgerError {
69    /// The underlying SQLite call failed.
70    #[error(transparent)]
71    Sqlite(#[from] rusqlite::Error),
72    /// The file predates per-item checkpoints, so its `checkpoints` table has
73    /// no `item_index` column and its rows cannot be interpreted against the
74    /// current schema.
75    #[error(
76        "this job's ledger predates per-item fan-out checkpoints and cannot be resumed \
77         — re-submit the job to start a fresh one"
78    )]
79    StaleSchema,
80}
81
82impl Ledger {
83    /// Open (creating if absent) a job's ledger at `path`, ensuring both
84    /// tables exist and `job_status` has its single row.
85    ///
86    /// `graph_fingerprint` is recorded only when `job_status` doesn't exist
87    /// yet (a fresh job) — reopening an existing ledger leaves the
88    /// originally-recorded fingerprint untouched, since comparing old vs.
89    /// new fingerprint is a resume endpoint's job, not `open`'s.
90    pub fn open(path: &Path, graph_fingerprint: &str) -> Result<Self, LedgerError> {
91        if let Some(parent) = path.parent() {
92            std::fs::create_dir_all(parent).ok();
93        }
94        let conn = Connection::open(path)?;
95        // A future reader (daemon startup scan, resume endpoint) may open
96        // its own connection to this same file while a running job's
97        // connection is mid-write. Without this, SQLite's default
98        // busy_timeout of 0 means that second connection gets an immediate
99        // SQLITE_BUSY instead of waiting briefly for the lock to clear.
100        conn.busy_timeout(std::time::Duration::from_secs(5))?;
101
102        // A table that already exists keeps its original shape under CREATE
103        // TABLE IF NOT EXISTS, so a ledger written before per-item
104        // checkpoints would survive to here and then fail at the first
105        // INSERT with "no such column: item_index" — an error that says
106        // nothing about what actually happened or what to do. Detect it here
107        // instead. An empty result means the table doesn't exist yet, which
108        // is just a fresh ledger.
109        let existing_columns: Vec<String> = {
110            let mut stmt = conn.prepare("PRAGMA table_info(checkpoints)")?;
111            let rows = stmt.query_map([], |r| r.get::<_, String>(1))?;
112            rows.collect::<rusqlite::Result<_>>()?
113        };
114        if !existing_columns.is_empty() && !existing_columns.iter().any(|c| c == "item_index") {
115            return Err(LedgerError::StaleSchema);
116        }
117
118        conn.execute_batch(
119            "CREATE TABLE IF NOT EXISTS checkpoints (
120                node_name    TEXT NOT NULL,
121                item_index   INTEGER NOT NULL DEFAULT -1,
122                status       TEXT NOT NULL,
123                output_json  TEXT,
124                error_text   TEXT,
125                completed_at TEXT NOT NULL,
126                PRIMARY KEY (node_name, item_index)
127             );
128             CREATE TABLE IF NOT EXISTS job_status (status TEXT NOT NULL, graph_fingerprint TEXT NOT NULL);
129             CREATE TABLE IF NOT EXISTS fanout_manifests (
130                node_name  TEXT PRIMARY KEY,
131                digest     TEXT NOT NULL,
132                item_count INTEGER NOT NULL
133             );",
134        )?;
135        let count: i64 = conn.query_row("SELECT COUNT(*) FROM job_status", [], |r| r.get(0))?;
136        if count == 0 {
137            conn.execute(
138                "INSERT INTO job_status (status, graph_fingerprint) VALUES ('running', ?1)",
139                [graph_fingerprint],
140            )?;
141        }
142        Ok(Self {
143            conn: Mutex::new(conn),
144            // Fan-out results are materialized beside the ledger. Deriving
145            // the directory from the path we were already given avoids
146            // threading a second, independently-computed notion of "this
147            // job's directory" through `run_job`, which could drift.
148            job_dir: path
149                .parent()
150                .unwrap_or_else(|| Path::new("."))
151                .to_path_buf(),
152        })
153    }

168    /// The recorded output of `node_name`, if it completed successfully.
169    /// `None` for a node that never ran, is still pending, or was skipped.
170    pub fn get_completed(&self, node_name: &str) -> rusqlite::Result<Option<serde_json::Value>> {
171        let result: Option<(String, Option<String>)> = match self.lock().query_row(
172            "SELECT status, output_json FROM checkpoints WHERE node_name = ?1 AND item_index = -1",
173            [node_name],
174            |r| Ok((r.get(0)?, r.get(1)?)),
175        ) {
176            Ok(row) => Some(row),
177            Err(rusqlite::Error::QueryReturnedNoRows) => None,
178            Err(e) => return Err(e),
179        };
180        match result {
181            Some((status, Some(json))) if status == "completed" => Ok(Some(
182                serde_json::from_str(&json).expect("ledger never stores invalid JSON"),
183            )),
184            _ => Ok(None),
185        }
186    }

188    /// Whether `node_name` was recorded as skipped (e.g. excluded by a
189    /// branch decision).
190    pub fn is_skipped(&self, node_name: &str) -> rusqlite::Result<bool> {
191        let status: Option<String> = match self.lock().query_row(
192            "SELECT status FROM checkpoints WHERE node_name = ?1 AND item_index = -1",
193            [node_name],
194            |r| r.get(0),
195        ) {
196            Ok(status) => Some(status),
197            Err(rusqlite::Error::QueryReturnedNoRows) => None,
198            Err(e) => return Err(e),
199        };
200        Ok(status.as_deref() == Some("skipped"))
201    }

203    /// Record `node_name` as completed with `output`, overwriting any prior
204    /// checkpoint for that node.
205    pub fn write_completed(
206        &self,
207        node_name: &str,
208        output: &serde_json::Value,
209    ) -> rusqlite::Result<()> {
210        self.lock().execute(
211            "INSERT OR REPLACE INTO checkpoints
212               (node_name, item_index, status, output_json, error_text, completed_at)
213             VALUES (?1, -1, 'completed', ?2, NULL, ?3)",
214            rusqlite::params![node_name, output.to_string(), now_marker()],
215        )?;
216        Ok(())
217    }

252    /// Record one fan-out item as having *concluded* in failure.
253    ///
254    /// Concluded is the operative word. An item still in flight when the
255    /// process died must leave no row at all, so that resume re-runs it —
256    /// whereas an item whose block genuinely returned `Fail` is recorded
257    /// here and never retried. That distinction is the entire basis of
258    /// fan-out resume semantics: it separates "this chunk is bad" from "we
259    /// were interrupted", without needing to ask which happened.
260    pub fn write_item_failed(
261        &self,
262        node_name: &str,
263        item_index: usize,
264        error: &str,
265    ) -> rusqlite::Result<()> {
266        self.lock().execute(
267            "INSERT OR REPLACE INTO checkpoints
268               (node_name, item_index, status, output_json, error_text, completed_at)
269             VALUES (?1, ?2, 'failed', NULL, ?3, ?4)",
270            rusqlite::params![node_name, item_index as i64, error, now_marker()],
271        )?;
272        Ok(())
273    }

277        &self,
278        node_name: &str,
279        item_index: usize,

299    /// Whether this item already concluded, either way — the resume check.
300    /// A concluded item is skipped; anything else is (re-)run.
301    pub fn item_concluded(&self, node_name: &str, item_index: usize) -> rusqlite::Result<bool> {
302        let count: i64 = self.lock().query_row(
303            "SELECT COUNT(*) FROM checkpoints WHERE node_name = ?1 AND item_index = ?2",
304            rusqlite::params![node_name, item_index as i64],
305            |r| r.get(0),
306        )?;
307        Ok(count > 0)
308    }

310    /// Every concluded item for `node_name`, in index order, as
311    /// `(index, output, error)` — exactly one of `output`/`error` is `Some`.
312    #[allow(clippy::type_complexity)]
313    pub fn concluded_items(
314        &self,
315        node_name: &str,
316    ) -> rusqlite::Result<Vec<(usize, Option<serde_json::Value>, Option<String>)>> {
317        let conn = self.lock();
318        let mut stmt = conn.prepare(
319            "SELECT item_index, status, output_json, error_text FROM checkpoints
320             WHERE node_name = ?1 AND item_index >= 0 ORDER BY item_index",
321        )?;
322        let rows = stmt.query_map([node_name], |r| {
323            let index: i64 = r.get(0)?;
324            let status: String = r.get(1)?;
325            let output: Option<String> = r.get(2)?;
326            let error: Option<String> = r.get(3)?;
327            Ok((
328                index as usize,
329                if status == "completed" {
330                    output.map(|j| {
331                        serde_json::from_str(&j).expect("ledger never stores invalid JSON")
332                    })
333                } else {
334                    None
335                },
336                if status == "failed" { error } else { None },
337            ))
338        })?;
339        rows.collect()
340    }

342    /// Record what `node_name` fanned out over, or verify it is unchanged.
343    ///
344    /// `Ok(Err(previous_digest))` means this node previously ran against a
345    /// *different* manifest. Item indices are only meaningful relative to one
346    /// specific manifest, so resuming would quietly pair recorded results
347    /// with entirely different inputs — no graph-level fingerprint can catch
348    /// an edit to the manifest file itself, which is why this exists.
349    ///
350    /// The outer `Result` is storage failure; the inner one is the verdict.
351    pub fn check_or_record_manifest(
352        &self,
353        node_name: &str,
354        digest: &str,
355        item_count: usize,
356    ) -> rusqlite::Result<Result<(), String>> {
357        let conn = self.lock();
358        let existing: Option<String> = match conn.query_row(
359            "SELECT digest FROM fanout_manifests WHERE node_name = ?1",
360            [node_name],
361            |r| r.get(0),
362        ) {
363            Ok(d) => Some(d),
364            Err(rusqlite::Error::QueryReturnedNoRows) => None,
365            Err(e) => return Err(e),
366        };
367        match existing {
368            Some(previous) if previous != digest => Ok(Err(previous)),
369            Some(_) => Ok(Ok(())),
370            None => {
371                conn.execute(
372                    "INSERT INTO fanout_manifests (node_name, digest, item_count)
373                     VALUES (?1, ?2, ?3)",
374                    rusqlite::params![node_name, digest, item_count as i64],
375                )?;
376                Ok(Ok(()))
377            }
378        }
379    }

381    /// The job's own terminal status. `Running` until [`Ledger::finish`] is
382    /// called.
383    pub fn job_status(&self) -> rusqlite::Result<LedgerJobStatus> {
384        let s: String = self

386            .query_row("SELECT status FROM job_status", [], |r| r.get(0))?;

390    /// Record the job's terminal status (e.g. `"completed"`, `"failed"`,
391    /// `"cancelled"`).
392    pub fn finish(&self, status: &str) -> rusqlite::Result<()> {
393        self.lock()
394            .execute("UPDATE job_status SET status = ?1", [status])?;
395        Ok(())
396    }

398    /// Lock the connection. The mutex is only ever held for the duration of
399    /// one synchronous rusqlite call, never across an `.await` — so a
400    /// poisoned lock can only mean a prior call panicked mid-query, an
401    /// exceptional situation worth propagating loudly rather than papering
402    /// over.
403    fn lock(&self) -> std::sync::MutexGuard<'_, Connection> {
404        self.conn.lock().expect("ledger connection mutex poisoned")
405    }

408/// The root directory jobs live under. Checks `$CUTTLEFISH_JOBS_HOME` first
409/// — set by `cuttlefish-run`'s project-scoping so a project's jobs/ledger
410/// state lives under `<project>/.cuttlefish/jobs` without also redirecting
411/// the (deliberately still-global) block catalog, which `$CUTTLEFISH_HOME`
412/// alone continues to control. Falls back to `$CUTTLEFISH_HOME/jobs` (or
413/// `~/.cuttlefish/jobs`) exactly as before when unset, so nothing about
414/// existing single-global-home behavior changes for a caller that never
415/// sets the new variable.
416pub fn jobs_root() -> Option<std::path::PathBuf> {
417    if let Ok(dir) = std::env::var("CUTTLEFISH_JOBS_HOME") {
418        return Some(std::path::PathBuf::from(dir));
419    }
420    crate::catalog::cuttlefish_home().map(|h| h.join("jobs"))
421}
