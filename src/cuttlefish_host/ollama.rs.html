1//! Inference served by a local [Ollama](https://ollama.com) instance.
2//!
3//! Ollama is the first real backend because it is the shortest path to a real
4//! model: it is already an HTTP server that streams, so this is a client rather
5//! than an embedding of llama.cpp. Embedding llama.cpp directly remains
6//! worthwhile later — it removes a process boundary and gives direct control of
7//! the KV cache — but it is a much larger commitment, and nothing above
8//! [`InferBackend`] can tell the difference.
9//!
10//! # The wire format
11//!
12//! `POST /api/generate` with `stream: true` responds with newline-delimited
13//! JSON — one object per token, then a final object carrying counts:
14//!
15//! ```text
16//! {"model":"llama3.2:1b","response":"Hello","done":false}
17//! {"model":"llama3.2:1b","response":" there","done":false}
18//! {"model":"llama3.2:1b","response":"","done":true,"done_reason":"stop",
19//!  "prompt_eval_count":28,"eval_count":3, ...}
20//! ```
21//!
22//! Token-at-a-time delivery is what makes a guest's early-stop verdict
23//! meaningful: the host can act on it while generation is still running rather
24//! than after the fact.
25//!
26//! # Pinned generation
27//!
28//! Plain targets such as `Ollama "qwen3.8:27b"` keep the original behaviour.
29//! A JSON target opts into a required digest pin and explicit generation controls:
30//!
31//! ```text
32//! model = Ollama "{\"model\":\"qwen3.8:27b\",\"expected_digest\":\"aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa\",\"seed\":42,\"temperature\":0.2,\"format\":\"json\",\"think\":false}";
33//! ```
34//!
35//! Replace the example digest with the actual 64-digit digest from
36//! [`GET /api/tags`](https://docs.ollama.com/api/tags). Each configured generation
37//! checks that mapping before and after inference and checks the response model.
38//! Usage records the requested `tag@sha256:digest`; successful inference means
39//! those checks passed. This is tag-mapping evidence, not atomic attestation of
40//! the weights served. Resuming stored results does not re-attest their identity.
41//! Keep the server and model files stable during a campaign.
42//!
43//! Supported [`generate`](https://docs.ollama.com/api/generate) controls are
44//! integer `seed`, nonnegative finite `temperature`, `format` (`"json"` or a JSON
45//! Schema object), and boolean `think`. Omitted controls retain Ollama defaults;
46//! `num_predict` always comes from the request budget. Unknown fields fail.
47//! Thinking and schema support remain model/server dependent; server errors
48//! propagate. A seed does not guarantee identical output across versions or
49//! hardware. Schema output still needs independent acceptance validation.
50

56/// Where Ollama listens when nothing says otherwise.
57pub const DEFAULT_HOST: &str = "http://localhost:11434";

59/// One line of Ollama's streaming response.
60///
61/// Deliberately partial: Ollama sends more fields than this (timings, and a
62/// `context` array that can run to thousands of integers). Naming only what is
63/// used keeps the deserializer from being coupled to fields the project does
64/// not care about, and avoids materializing that context array on every call.
65#[derive(Debug, Deserialize)]
66struct Chunk {
67    #[serde(default)]
68    model: Option<String>,
69    #[serde(default)]
70    response: String,
71    #[serde(default)]
72    done: bool,
73    #[serde(default)]
74    prompt_eval_count: Option<u32>,
75    #[serde(default)]
76    eval_count: Option<u32>,
77    /// Present when Ollama itself reports a problem — an unknown model, say.
78    #[serde(default)]
79    error: Option<String>,
80}

82/// Serves inference from a local Ollama instance.
83pub struct OllamaBackend {
84    client: reqwest::Client,
85    host: String,
86    model: String,
87    config: Option<GenerationConfig>,
88}

90#[derive(Debug, Deserialize)]
91#[serde(deny_unknown_fields)]
92struct GenerationConfig {
93    model: String,
94    expected_digest: String,
95    seed: Option<i64>,
96    temperature: Option<f64>,
97    format: Option<serde_json::Value>,
98    think: Option<bool>,
99}
100
101fn canonical_digest(value: &str) -> anyhow::Result<String> {
102    let value = value.strip_prefix("sha256:").unwrap_or(value);
103    anyhow::ensure!(
104        value.len() == 64 && value.bytes().all(|c| c.is_ascii_hexdigit()),
105        "Ollama digest must contain exactly 64 hexadecimal digits"
106    );
107    Ok(value.to_ascii_lowercase())
108}

110impl OllamaBackend {
111    /// Target `model` on the Ollama instance at `host`.
112    ///
113    /// `model` is the name as Ollama knows it, `:tag` included.
114    pub fn new(host: impl Into<String>, model: impl Into<String>) -> Self {
115        Self {
116            client: reqwest::Client::new(),
117            // A trailing slash here would produce `//api/generate`, which Ollama
118            // rejects — easy to introduce via an environment variable.
119            host: host.into().trim_end_matches('/').to_string(),
120            model: model.into(),
121            config: None,
122        }
123    }
124
125    /// Build a legacy model tag or a validated JSON generation target.
126    /// No network requests occur until inference.
127    pub fn from_target(host: impl Into<String>, target: &str) -> anyhow::Result<Self> {
128        anyhow::ensure!(
129            !target.trim().is_empty(),
130            "an Ollama model name is required"
131        );
132        if !target.trim_start().starts_with('{') {
133            return Ok(Self::new(host, target));
134        }
135        let mut config: GenerationConfig = serde_json::from_str(target)
136            .map_err(|e| anyhow::anyhow!("invalid Ollama generation configuration: {e}"))?;
137        // Exact tag names avoid silently pinning one spelling while the server
138        // resolves a different default tag in the generation request.
139        anyhow::ensure!(
140            !config.model.chars().any(char::is_whitespace)
141                && config.model.rsplit('/').next().is_some_and(|last| {
142                    last.split_once(':')
143                        .is_some_and(|(name, tag)| !name.is_empty() && !tag.is_empty())
144                }),
145            "configured Ollama model requires an explicit name:tag"
146        );
147        config.expected_digest = canonical_digest(&config.expected_digest)?;
148        if let Some(temperature) = config.temperature {
149            anyhow::ensure!(
150                temperature.is_finite() && temperature >= 0.0,
151                "Ollama temperature must be finite and nonnegative"
152            );
153        }
154        if let Some(format) = &config.format {
155            if format.is_object() {
156                jsonschema::validator_for(format)
157                    .map_err(|e| anyhow::anyhow!("invalid Ollama format schema: {e}"))?;
158            } else {
159                anyhow::ensure!(
160                    format.as_str() == Some("json"),
161                    "Ollama format must be 'json' or a JSON Schema object"
162                );
163            }
164        }
165        let mut backend = Self::new(host, config.model.clone());
166        backend.config = Some(config);
167        Ok(backend)
168    }
169
170    async fn verify_digest(&self, config: &GenerationConfig) -> anyhow::Result<()> {
171        let response = self
172            .client
173            .get(format!("{}/api/tags", self.host))
174            .send()
175            .await?
176            .error_for_status()?;
177        let body: serde_json::Value = response.json().await?;
178        let models = body
179            .get("models")
180            .and_then(serde_json::Value::as_array)
181            .ok_or_else(|| anyhow::anyhow!("Ollama /api/tags response has no models array"))?;
182        let matches: Vec<_> = models
183            .iter()
184            .filter(|entry| {
185                entry.get("name").and_then(serde_json::Value::as_str) == Some(self.model.as_str())
186            })
187            .collect();
188        anyhow::ensure!(
189            matches.len() == 1,
190            "Ollama /api/tags must contain exactly one entry named {} (found {})",
191            self.model,
192            matches.len()
193        );
194        let digest = matches[0]
195            .get("digest")
196            .and_then(serde_json::Value::as_str)
197            .ok_or_else(|| anyhow::anyhow!("Ollama tag {} has no digest", self.model))?;
198        let digest = canonical_digest(digest)?;
199        anyhow::ensure!(
200            digest == config.expected_digest,
201            "Ollama digest mismatch for {}: expected {}, observed {}",
202            self.model,
203            config.expected_digest,
204            digest
205        );
206        Ok(())
207    }
208
209    /// Read `OLLAMA_HOST` if set, otherwise [`DEFAULT_HOST`].
210    ///
211    /// Named for the variable Ollama's own tooling uses, so an operator who has
212    /// already pointed their CLI at a non-default instance does not have to
213    /// configure this separately.
214    pub fn host_from_env() -> String {
215        std::env::var("OLLAMA_HOST").unwrap_or_else(|_| DEFAULT_HOST.to_string())
216    }

219#[async_trait]
220impl InferBackend for OllamaBackend {
221    async fn infer(
222        &self,
223        req: InferRequest<'_>,
224        on_token: &mut (dyn for<'t> FnMut(&'t str) -> bool + Send),
225    ) -> anyhow::Result<InferResult> {
226        // Ollama takes images as base64 strings alongside the prompt. Whether
227        // the *model* can use them is Ollama's business — asking a text-only
228        // model produces a clear error from it, which is more useful than a
229        // guess made here from a model name.
230        let images: Vec<String> = req
231            .images
232            .iter()
233            .map(|bytes| {
234                use base64::Engine;
235                base64::engine::general_purpose::STANDARD.encode(bytes)
236            })
237            .collect();
238
239        let mut payload = serde_json::json!({
240            "model": self.model,
241            "prompt": req.prompt,
242            "stream": true,
243            "images": images,
244            "options": { "num_predict": req.max_tokens },
245        });
246        if let Some(config) = &self.config {
247            self.verify_digest(config).await?;
248            if let Some(seed) = config.seed {
249                payload["options"]["seed"] = seed.into();
250            }
251            if let Some(temperature) = config.temperature {
252                payload["options"]["temperature"] = temperature.into();
253            }
254            if let Some(format) = &config.format {
255                payload["format"] = format.clone();
256            }
257            if let Some(think) = config.think {
258                payload["think"] = think.into();
259            }
260        }
261        let response = self
262            .client
263            .post(format!("{}/api/generate", self.host))
264            .json(&payload)
265            .send()
266            .await
267            .map_err(|e| {
268                // The overwhelmingly likely cause is that Ollama is not running,
269                // and reqwest's own message does not say so.
270                anyhow::anyhow!(
271                    "could not reach Ollama at {} ({e}). Is it running? \
272                     Set OLLAMA_HOST to point elsewhere.",
273                    self.host
274                )
275            })?;
276
277        if !response.status().is_success() {
278            let status = response.status();
279            let body = response.text().await.unwrap_or_default();
280            anyhow::bail!("Ollama returned {status}: {body}");
281        }
282
283        let mut text = String::new();
284        let mut tokens_in = 0;
285        let mut tokens_out = 0;
286        let mut terminal = false;
287        let mut stopped = false;
288
289        // Responses are newline-delimited JSON, and a chunk boundary need not
290        // fall on a newline — a line can arrive split across two chunks, and one
291        // chunk can carry several lines. Buffering and splitting on '\n' is what
292        // makes this correct rather than usually-correct.
293        let mut stream = response.bytes_stream();
294        let mut buf = String::new();
295
296        'outer: while let Some(chunk) = stream.next().await {
297            buf.push_str(&String::from_utf8_lossy(&chunk?));
298
299            while let Some(newline) = buf.find('\n') {
300                // Take everything before the newline; leave the remainder in the
301                // buffer as the start of the next line.
302                let line: String = buf.drain(..=newline).collect();
303                let line = line.trim();
304                if line.is_empty() {
305                    continue;
306                }
307
308                let chunk: Chunk = serde_json::from_str(line).map_err(|e| {
309                    anyhow::anyhow!("malformed response from Ollama: {e} in {line}")
310                })?;
311
312                if let Some(error) = chunk.error {
313                    anyhow::bail!("Ollama error: {error}");
314                }
315                if self.config.is_some() {
316                    anyhow::ensure!(
317                        chunk.model.as_deref() == Some(self.model.as_str()),
318                        "Ollama response model {:?} does not match configured {}",
319                        chunk.model,
320                        self.model
321                    );
322                }
323
324                if !chunk.response.is_empty() {
325                    text.push_str(&chunk.response);
326                    tokens_out += 1;
327                    if !on_token(&chunk.response) {
328                        stopped = true;
329                        // The guest asked to stop. Dropping the stream closes
330                        // the connection, which is how Ollama learns to stop
331                        // generating — there is no separate cancel call.
332                        break 'outer;
333                    }
334                }
335
336                if chunk.done {
337                    terminal = true;
338                    // The final message carries authoritative counts; prefer
339                    // them over the tokens counted above, which only sees
340                    // non-empty responses.
341                    tokens_in = chunk.prompt_eval_count.unwrap_or(0);
342                    tokens_out = chunk.eval_count.unwrap_or(tokens_out);
343                    break 'outer;
344                }
345            }
346        }
347
348        if let Some(config) = &self.config {
349            anyhow::ensure!(
350                terminal || stopped,
351                "Ollama stream ended without a terminal response"
352            );
353            self.verify_digest(config).await?;
354        }
355
356        Ok(InferResult {
357            text,
358            tokens_in,
359            tokens_out,
360        })
361    }

363    fn model_name(&self) -> String {
364        match &self.config {
365            Some(config) => format!("{}@sha256:{}", self.model, config.expected_digest),
366            None => self.model.clone(),
367        }
368    }
369
370    /// True, but the honest claim is narrower than the signature allows:
371    /// whether images work depends on the *model*, not on this backend, and
372    /// this type does not know which model it is pointed at until a request is
373    /// made.
374    ///
375    /// Forwarding is nonetheless the better behaviour, because Ollama answers
376    /// the question authoritatively and specifically — HTTP 400 with
377    /// "Multimodal data provided, but model does not support multimodal
378    /// requests" (verified against llama3.2:1b). Guessing here from a model
379    /// name would produce a worse message and a new way to be wrong, since the
380    /// set of vision models changes without this code changing.
381    fn supports_images(&self) -> bool {
382        true
383    }

385    async fn embed(&self, texts: &[String]) -> anyhow::Result<Vec<Vec<f32>>> {
386        anyhow::ensure!(self.config.is_none(),
387            "Ollama generation configuration is not an embedding target; use a plain embedding model tag");
388        if texts.is_empty() {
389            return Ok(Vec::new());
390        }
391        // `/api/embed` (not the older `/api/embeddings`) because it takes an
392        // array and answers with one vector per input. The legacy endpoint
393        // is one text per request, which on a corpus is the difference
394        // between minutes and hours.
395        let response = self
396            .client
397            .post(format!("{}/api/embed", self.host))
398            .json(&serde_json::json!({ "model": self.model, "input": texts }))
399            .send()
400            .await
401            .map_err(|e| anyhow::anyhow!("embedding via {}: {e}", self.host))?;
402
403        let status = response.status();
404        let body = response
405            .text()
406            .await
407            .map_err(|e| anyhow::anyhow!("reading the embedding response: {e}"))?;
408        if !status.is_success() {
409            anyhow::bail!("embedding with `{}`: {status}: {body}", self.model);
410        }
411
412        let parsed: serde_json::Value = serde_json::from_str(&body)
413            .map_err(|e| anyhow::anyhow!("embedding response was not JSON: {e}: {body}"))?;
414        let rows = parsed
415            .get("embeddings")
416            .and_then(|v| v.as_array())
417            .ok_or_else(|| {
418                anyhow::anyhow!("embedding response had no `embeddings` array: {body}")
419            })?;
420
421        // One vector per input, in order. A mismatch means the rows can no
422        // longer be paired with their texts, and pairing them wrongly is
423        // worse than failing: every downstream similarity would be against
424        // the wrong content.
425        if rows.len() != texts.len() {
426            anyhow::bail!(
427                "asked `{}` for {} embeddings and got {} — refusing to pair vectors with \
428                 texts by guesswork",
429                self.model,
430                texts.len(),
431                rows.len()
432            );
433        }
434
435        let mut out = Vec::with_capacity(rows.len());
436        for (i, row) in rows.iter().enumerate() {
437            let vector: Vec<f32> = row
438                .as_array()
439                .ok_or_else(|| anyhow::anyhow!("embedding {i} was not an array: {row}"))?
440                .iter()
441                .map(|v| v.as_f64().unwrap_or(f64::NAN) as f32)
442                .collect();
443            if vector.is_empty() || vector.iter().any(|v| v.is_nan()) {
444                anyhow::bail!(
445                    "embedding {i} from `{}` is empty or contains non-numbers; storing it \
446                     would poison every similarity computed against it",
447                    self.model
448                );
449            }
450            out.push(vector);
451        }
452        Ok(out)
453    }
454
455    fn supports_embeddings(&self) -> bool {
456        self.config.is_none()
457    }
458}

460/// Builds [`OllamaBackend`], registered as the `ollama` provider.
461pub struct OllamaFactory;
462
463impl crate::backend::BackendFactory for OllamaFactory {
464    fn provider(&self) -> &'static str {
465        "ollama"
466    }
467
468    fn describe(&self) -> &'static str {
469        "a local Ollama instance; target is a model tag or a JSON generation configuration with expected_digest"
470    }
471
472    fn build(&self, target: &str) -> anyhow::Result<std::sync::Arc<dyn InferBackend>> {
473        if target.is_empty() {
474            anyhow::bail!("an Ollama model name is required, e.g. `llama3.2:1b`");
475        }
476        // Reachability is deliberately not checked here — see `BackendFactory`.
477        Ok(std::sync::Arc::new(OllamaBackend::from_target(
478            OllamaBackend::host_from_env(),
479            target,
480        )?))
481    }
