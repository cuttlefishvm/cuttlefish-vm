1//! Serving the API over a machine-local, OS-permission-gated transport: a
2//! unix domain socket on unix, a named pipe on Windows.
3//!
4//! Both are deliberate choices over a TCP listener. The daemon's access
5//! control *is* the endpoint's own permissions — it executes wasm and drives
6//! local models, so "any process on this machine may connect" is a materially
7//! weaker posture than "whoever can open this socket file may connect". A
8//! loopback TCP port has no such gate and would need an auth token bolted on
9//! just to get back to where we already are. Named pipes are the Windows
10//! analogue of the unix socket, not a workaround for the lack of one.
11
12use axum::Router;
13use std::path::Path;
14
15/// Serve `app` on `endpoint` until `shutdown` resolves.
16///
17/// `endpoint` is a socket path on unix and a pipe name (`\\.\pipe\...`) on
18/// Windows; see [`cuttlefish_core::endpoint::default_endpoint`]. `shutdown`
19/// is the portable stand-in for a `SIGTERM` handler — Windows has no signal
20/// this daemon could install one for, so graceful stop is instead triggered
21/// by the `POST /shutdown` route resolving this future; see `api::AppState`.
22pub async fn serve(
23    app: Router,
24    endpoint: &Path,
25    shutdown: impl std::future::Future<Output = ()> + Send + 'static,
26) -> anyhow::Result<()> {
27    #[cfg(unix)]
28    {
29        serve_unix(app, endpoint, shutdown).await
30    }
31    #[cfg(windows)]
32    {
33        serve_named_pipe(app, endpoint, shutdown).await
34    }
35}
36
37/// Serve `app` on a unix socket at `sock_path` until `shutdown` resolves.
38///
39/// axum 0.8's `serve` accepts any listener, so this is a handful of lines. On
40/// 0.7 it took a concrete `TcpListener` and a unix socket needed a hand-rolled
41/// hyper accept loop over `hyper-util` and `tower`; if you find yourself
42/// reintroducing that, check the axum version first.
43#[cfg(unix)]
44pub async fn serve_unix(
45    app: Router,
46    sock_path: &Path,
47    shutdown: impl std::future::Future<Output = ()> + Send + 'static,
48) -> anyhow::Result<()> {
49    use tokio::net::UnixListener;
50
51    // A stale socket file from a previous run makes bind() fail with EADDRINUSE
52    // even though nothing is listening on it.
53    let _ = std::fs::remove_file(sock_path);
54
55    let listener = UnixListener::bind(sock_path)?;
56    eprintln!("cuttlefishd listening on {}", sock_path.display());
57    axum::serve(listener, app)
58        .with_graceful_shutdown(shutdown)
59        .await?;
60    Ok(())
61}

63/// Serve `app` on a Windows named pipe until `shutdown` resolves.
64#[cfg(windows)]
65pub async fn serve_named_pipe(
66    app: Router,
67    pipe_name: &Path,
68    shutdown: impl std::future::Future<Output = ()> + Send + 'static,
69) -> anyhow::Result<()> {
70    let listener = windows_pipe::NamedPipeListener::bind(pipe_name)?;
71    eprintln!("cuttlefishd listening on {}", pipe_name.display());
72    axum::serve(listener, app)
73        .with_graceful_shutdown(shutdown)
74        .await?;
75    Ok(())
76}

78/// A named pipe presented as an [`axum::serve::Listener`].
79///
80/// Unlike a socket, a named pipe server does not multiplex: one server
81/// *instance* serves exactly one client. To stay continuously available you
82/// must have the next instance created before handing the current one off, or
83/// a client connecting in that window is refused. This module exists to own
84/// that rotation so nothing above it has to know about it.
85#[cfg(windows)]
86mod windows_pipe {
87    use std::ffi::OsString;
88    use std::io;
89    use std::path::Path;
90    use tokio::net::windows::named_pipe::{NamedPipeServer, ServerOptions};
91
92    pub struct NamedPipeListener {
93        name: OsString,
94        /// The instance currently waiting for a client. Always present:
95        /// `accept` creates its replacement before yielding the connected one.
96        pending: NamedPipeServer,
97    }
98
99    impl NamedPipeListener {
100        pub fn bind(name: &Path) -> io::Result<Self> {
101            let name = name.as_os_str().to_os_string();
102            // `first_pipe_instance` makes a second daemon on the same name fail
103            // loudly here rather than silently taking half the connections.
104            let pending = ServerOptions::new()
105                .first_pipe_instance(true)
106                .create(&name)?;
107            Ok(Self { name, pending })
108        }
109    }
110
111    impl axum::serve::Listener for NamedPipeListener {
112        type Io = NamedPipeServer;
113        /// A pipe connection has no peer address to report. axum only requires
114        /// this to be `Send`.
115        type Addr = ();
116
117        async fn accept(&mut self) -> (Self::Io, Self::Addr) {
118            loop {
119                // Waits until a client opens the pipe.
120                if let Err(e) = self.pending.connect().await {
121                    eprintln!("cuttlefishd: named pipe connect failed: {e}");
122                    continue;
123                }
124
125                // Create the *next* instance before yielding this one, so the
126                // pipe name is never momentarily unavailable.
127                match ServerOptions::new().create(&self.name) {
128                    Ok(next) => {
129                        let connected = std::mem::replace(&mut self.pending, next);
130                        return (connected, ());
131                    }
132                    Err(e) => {
133                        // Yielding this connection would leave no listener
134                        // behind, so drop it and keep serving instead. The
135                        // trait requires accept to handle its own errors.
136                        eprintln!("cuttlefishd: could not create the next pipe instance: {e}");
137                        self.pending.disconnect().ok();
138                    }
139                }
140            }
141        }
142
143        fn local_addr(&self) -> io::Result<Self::Addr> {
144            Ok(())
145        }
146    }
147}
