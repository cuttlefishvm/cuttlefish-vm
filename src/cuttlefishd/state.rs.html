1//! Job bookkeeping: status, results, and the event log.
2//!
3//! # Why events are logged and not merely broadcast
4//!
5//! `tokio::sync::broadcast` has no replay for late subscribers. `send` returns
6//! early when the receiver count is zero and does not even write to the ring
7//! buffer, and `subscribe` starts a receiver at the current tail.
8//!
9//! That is fatal for the obvious client shape: submit a job, then connect to its
10//! event stream. Those are two round trips, and a fast job finishes during the
11//! gap — so a broadcast-only design delivers *nothing*, reliably, and looks like
12//! a streaming bug rather than a design one.
13//!
14//! So every event goes into a per-job log as well as the live channel, and
15//! subscribing hands back the backlog together with a receiver. Finished
16//! envelopes are retained for the same reason: a dropped connection must not be
17//! able to lose a result that the daemon already computed.
18
19use cuttlefish_abi::{Envelope, JobStatus};
20use std::collections::HashMap;
21use std::sync::Arc;
22use tokio::sync::{broadcast, Mutex};
23use tokio_util::sync::CancellationToken;
24
25/// How many live events a subscriber may fall behind before it is dropped.
26///
27/// Falling behind is survivable rather than fatal: the log holds everything, so
28/// a lagging subscriber that reconnects catches up from the backlog.
29const BROADCAST_CAPACITY: usize = 256;
30
31/// One job's state.
32#[derive(Clone)]
33pub struct Job {
34    /// Unique identifier, minted at submission.
35    pub id: String,
36    /// Lifecycle state.
37    pub status: JobStatus,
38    /// Present once the job reaches a terminal state.
39    pub envelope: Option<Envelope>,
40    /// Cancels the running job; see the runner.
41    pub cancel: CancellationToken,
42    /// Live events, for subscribers already attached.
43    pub events: broadcast::Sender<String>,
44    /// Everything ever published for this job, so a late subscriber catches up.
45    ///
46    /// Unbounded, which is fine at this slice's scale — a job emits tens of
47    /// events. A cap belongs here once jobs run long enough for it to matter.
48    pub log: Vec<String>,
49}
50
51impl Job {
52    /// Create a queued job with a fresh event channel.
53    pub fn new(id: String, cancel: CancellationToken) -> Self {
54        let (events, _) = broadcast::channel(BROADCAST_CAPACITY);
55        Self {
56            id,
57            status: JobStatus::Running,
58            envelope: None,
59            cancel,
60            events,
61            log: Vec::new(),
62        }
63    }
64
65    /// Reconstruct a job discovered mid-flight in its ledger at daemon
66    /// startup — not actually running (there is no task backing it), just
67    /// recorded as having been running when the process last died.
68    ///
69    /// The `CancellationToken` is fresh and never cancelled: nothing is
70    /// executing for this job in this process, so there is nothing for it to
71    /// cancel. `envelope` stays `None` — there is nothing to show yet; a
72    /// resume endpoint is what eventually gives this job a real envelope.
73    pub fn interrupted(id: String) -> Self {
74        let (events, _) = broadcast::channel(BROADCAST_CAPACITY);
75        Self {
76            id,
77            status: JobStatus::Interrupted,
78            envelope: None,
79            cancel: CancellationToken::new(),
80            events,
81            log: Vec::new(),
82        }
83    }
84}
85
86/// Every job this daemon knows about.
87#[derive(Clone, Default)]
88pub struct JobStore {
89    jobs: Arc<Mutex<HashMap<String, Job>>>,
90}
91
92impl JobStore {
93    /// Record a new job.
94    pub async fn insert(&self, job: Job) {
95        self.jobs.lock().await.insert(job.id.clone(), job);
96    }
97
98    /// Fetch a snapshot of a job, if it exists.
99    pub async fn get(&self, id: &str) -> Option<Job> {
100        self.jobs.lock().await.get(id).cloned()
101    }
102
103    /// Append to the replay log and fan out to live subscribers.
104    ///
105    /// Both happen under one lock so an event can never be simultaneously missed
106    /// by a subscriber and absent from the log. A failing `send` just means
107    /// nobody is attached yet, which the log covers.
108    pub async fn publish(&self, id: &str, payload: String) {
109        if let Some(job) = self.jobs.lock().await.get_mut(id) {
110            job.log.push(payload.clone());
111            let _ = job.events.send(payload);
112        }
113    }
114
115    /// Snapshot the backlog and attach a live receiver.
116    ///
117    /// Both under the *same* lock guard. Splitting them would reopen the gap
118    /// this exists to close: an event published between the snapshot and the
119    /// subscribe would appear in neither.
120    pub async fn subscribe(&self, id: &str) -> Option<(Vec<String>, broadcast::Receiver<String>)> {
121        let jobs = self.jobs.lock().await;
122        let job = jobs.get(id)?;
123        Some((job.log.clone(), job.events.subscribe()))
124    }
125
126    /// Record a job's terminal envelope.
127    pub async fn finish(&self, id: &str, envelope: Envelope) {
128        if let Some(job) = self.jobs.lock().await.get_mut(id) {
129            job.status = envelope.status;
130            job.envelope = Some(envelope);
131        }
132    }
133
134    /// Record a job discovered mid-flight in its ledger at daemon startup;
135    /// see [`Job::interrupted`].
136    pub async fn mark_interrupted(&self, id: String) {
137        self.insert(Job::interrupted(id)).await;
138    }
139
140    /// Snapshot every known job, in the same `{job_id, status, envelope}`
141    /// shape `get_job` returns for one — this is what makes an `Interrupted`
142    /// job (surfaced only via the startup scan, never streamed) actually
143    /// discoverable: an agent lists jobs, sees one sitting at `Interrupted`,
144    /// and only then decides whether to call resume.
145    pub async fn list(&self) -> Vec<serde_json::Value> {
146        self.jobs
147            .lock()
148            .await
149            .values()
150            .map(|job| {
151                serde_json::json!({
152                    "job_id": job.id,
153                    "status": job.status,
154                    "envelope": job.envelope,
155                })
156            })
157            .collect()
158    }
159
160    /// Request cancellation. Returns whether the job existed.
161    ///
162    /// Best-effort by nature: a job that has already finished is reported as
163    /// cancelled-successfully because the caller's intent — "do not keep running
164    /// this" — is already satisfied.
165    ///
166    /// A job discovered `Interrupted` at startup is a different case: its
167    /// `CancellationToken` is fresh and wired to nothing (see
168    /// [`Job::interrupted`]), so calling `.cancel()` on it would silently do
169    /// nothing and leave `status` stuck at `Interrupted` forever, while still
170    /// reporting success — unlike an already-terminal job, its final
171    /// disposition is genuinely unresolved (it could still be resumed), so
172    /// "intent already satisfied" doesn't apply. Instead it's transitioned
173    /// directly to `Cancelled` here. That conveniently also closes the loop
174    /// for resume: a resume check against `status == Interrupted` correctly
175    /// refuses a job cancelled this way, with no extra plumbing needed.
176    ///
177    /// The on-disk ledger is also updated to match: without this, its
178    /// `job_status` table would still say `'running'`, and a subsequent
179    /// daemon restart's startup scan would re-read that, silently re-mark
180    /// the job `Interrupted`, and discard this cancel decision entirely —
181    /// the ledger is the authoritative durability record, so it must not be
182    /// allowed to diverge from the in-memory view like that.
183    pub async fn cancel(&self, id: &str) -> bool {
184        let mut jobs = self.jobs.lock().await;
185        match jobs.get_mut(id) {
186            Some(job) if job.status == JobStatus::Interrupted => {
187                job.status = JobStatus::Cancelled;
188                // Persist to the ledger too — otherwise a restart before
189                // this job's final disposition is otherwise resolved would
190                // silently re-mark it Interrupted, discarding this cancel
191                // decision. Best effort: if the ledger write fails, the
192                // in-memory transition above still holds for this process's
193                // lifetime; only a subsequent restart would be affected,
194                // which is a narrower, more acceptable failure window than
195                // not trying at all.
196                if let Some(jobs_root) = cuttlefish_host::ledger::jobs_root() {
197                    let ledger_path = jobs_root.join(id).join("ledger.sqlite");
198                    // Any existing fingerprint is preserved by
199                    // `Ledger::open` (only inserted on first creation) — the
200                    // value passed here is irrelevant for an
201                    // already-existing ledger, which this always is by the
202                    // time a job can be cancelled.
203                    if let Ok(ledger) = cuttlefish_host::ledger::Ledger::open(&ledger_path, "") {
204                        let _ = ledger.finish("cancelled");
205                    }
206                }
207                true
208            }
209            Some(job) => {
210                job.cancel.cancel();
211                true
212            }
213            None => false,
214        }
215    }
216
217    /// Atomically transition a job from `Interrupted` to `Running`, if and
218    /// only if it's still `Interrupted` at the moment this runs — the guard
219    /// against two concurrent `POST /jobs/{id}/resume` calls both winning and
220    /// spawning duplicate `run_job` executions against the same ledger.
221    ///
222    /// Returns the job (with its now-updated status) if this call won the
223    /// race, or `None` if another call already claimed it first (or the job
224    /// doesn't exist). Callers must run every other pre-flight check first —
225    /// this method exists to be the *last* thing before `tokio::spawn`, not a
226    /// replacement for the earlier checks; see `resume_job`'s ordering note.
227    pub async fn try_start_resume(&self, id: &str) -> Option<Job> {
228        let mut jobs = self.jobs.lock().await;
229        match jobs.get_mut(id) {
230            Some(job) if job.status == JobStatus::Interrupted => {
231                job.status = JobStatus::Running;
232                Some(job.clone())
233            }
234            _ => None,
235        }
236    }
237}
