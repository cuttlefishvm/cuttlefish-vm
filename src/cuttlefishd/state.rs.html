1//! Job bookkeeping: status, results, and the event log.
2//!
3//! # Why events are logged and not merely broadcast
4//!
5//! `tokio::sync::broadcast` has no replay for late subscribers. `send` returns
6//! early when the receiver count is zero and does not even write to the ring
7//! buffer, and `subscribe` starts a receiver at the current tail.
8//!
9//! That is fatal for the obvious client shape: submit a job, then connect to its
10//! event stream. Those are two round trips, and a fast job finishes during the
11//! gap — so a broadcast-only design delivers *nothing*, reliably, and looks like
12//! a streaming bug rather than a design one.
13//!
14//! So every event goes into a per-job log as well as the live channel, and
15//! subscribing hands back the backlog together with a receiver. Finished
16//! envelopes are retained for the same reason: a dropped connection must not be
17//! able to lose a result that the daemon already computed.
18
19use cuttlefish_abi::{Envelope, JobStatus};
20use std::collections::HashMap;
21use std::sync::Arc;
22use tokio::sync::{broadcast, Mutex};
23use tokio_util::sync::CancellationToken;
24
25/// How many live events a subscriber may fall behind before it is dropped.
26///
27/// Falling behind is survivable rather than fatal: the log holds everything, so
28/// a lagging subscriber that reconnects catches up from the backlog.
29const BROADCAST_CAPACITY: usize = 256;
30
31/// One job's state.
32#[derive(Clone)]
33pub struct Job {
34    /// Unique identifier, minted at submission.
35    pub id: String,
36    /// Lifecycle state.
37    pub status: JobStatus,
38    /// Present once the job reaches a terminal state.
39    pub envelope: Option<Envelope>,
40    /// Cancels the running job; see the runner.
41    pub cancel: CancellationToken,
42    /// Live events, for subscribers already attached.
43    pub events: broadcast::Sender<String>,
44    /// Everything ever published for this job, so a late subscriber catches up.
45    ///
46    /// Unbounded, which is fine at this slice's scale — a job emits tens of
47    /// events. A cap belongs here once jobs run long enough for it to matter.
48    pub log: Vec<String>,
49}
50
51impl Job {
52    /// Checkpoint work can resume after interruption or a failed publication.
53    /// Other terminal failures require a new job rather than implicit retries.
54    pub fn can_resume(&self) -> bool {
55        self.status == JobStatus::Interrupted
56            || (self.status == JobStatus::Failed
57                && self
58                    .envelope
59                    .as_ref()
60                    .and_then(|e| e.error.as_ref())
61                    .is_some_and(|e| {
62                        e.code == cuttlefish_abi::error_codes::WAREHOUSE_PUBLICATION_FAILED
63                    }))
64    }
65
66    /// Create a queued job with a fresh event channel.
67    pub fn new(id: String, cancel: CancellationToken) -> Self {
68        let (events, _) = broadcast::channel(BROADCAST_CAPACITY);
69        Self {
70            id,
71            status: JobStatus::Running,
72            envelope: None,
73            cancel,
74            events,
75            log: Vec::new(),
76        }
77    }
78
79    /// Reconstruct a job discovered mid-flight in its ledger at daemon
80    /// startup — not actually running (there is no task backing it), just
81    /// recorded as having been running when the process last died.
82    ///
83    /// The `CancellationToken` is fresh and never cancelled: nothing is
84    /// executing for this job in this process, so there is nothing for it to
85    /// cancel. `envelope` stays `None` — there is nothing to show yet; a
86    /// resume endpoint is what eventually gives this job a real envelope.
87    pub fn interrupted(id: String) -> Self {
88        let (events, _) = broadcast::channel(BROADCAST_CAPACITY);
89        Self {
90            id,
91            status: JobStatus::Interrupted,
92            envelope: None,
93            cancel: CancellationToken::new(),
94            events,
95            log: Vec::new(),
96        }
97    }
98}

100/// Every job this daemon knows about.
101#[derive(Clone, Default)]
102pub struct JobStore {
103    jobs: Arc<Mutex<HashMap<String, Job>>>,
104}
105
106impl JobStore {
107    /// Record a new job.
108    pub async fn insert(&self, job: Job) {
109        self.jobs.lock().await.insert(job.id.clone(), job);
110    }
111
112    /// Fetch a snapshot of a job, if it exists.
113    pub async fn get(&self, id: &str) -> Option<Job> {
114        self.jobs.lock().await.get(id).cloned()
115    }
116
117    /// Append to the replay log and fan out to live subscribers.
118    ///
119    /// Both happen under one lock so an event can never be simultaneously missed
120    /// by a subscriber and absent from the log. A failing `send` just means
121    /// nobody is attached yet, which the log covers.
122    pub async fn publish(&self, id: &str, payload: String) {
123        if let Some(job) = self.jobs.lock().await.get_mut(id) {
124            job.log.push(payload.clone());
125            let _ = job.events.send(payload);
126        }
127    }
128
129    /// Snapshot the backlog and attach a live receiver.
130    ///
131    /// Both under the *same* lock guard. Splitting them would reopen the gap
132    /// this exists to close: an event published between the snapshot and the
133    /// subscribe would appear in neither.
134    pub async fn subscribe(&self, id: &str) -> Option<(Vec<String>, broadcast::Receiver<String>)> {
135        let jobs = self.jobs.lock().await;
136        let job = jobs.get(id)?;
137        Some((job.log.clone(), job.events.subscribe()))
138    }
139
140    /// Record a job's terminal envelope.
141    pub async fn finish(&self, id: &str, envelope: Envelope) {
142        if let Some(job) = self.jobs.lock().await.get_mut(id) {
143            job.status = envelope.status;
144            job.envelope = Some(envelope);
145        }
146    }
147
148    /// Record a job discovered mid-flight in its ledger at daemon startup;
149    /// see [`Job::interrupted`].
150    pub async fn mark_interrupted(&self, id: String) {
151        self.insert(Job::interrupted(id)).await;
152    }
153
154    /// Snapshot every known job, in the same `{job_id, status, envelope}`
155    /// shape `get_job` returns for one — this is what makes an `Interrupted`
156    /// job (surfaced only via the startup scan, never streamed) actually
157    /// discoverable: an agent lists jobs, sees one sitting at `Interrupted`,
158    /// and only then decides whether to call resume.
159    pub async fn list(&self) -> Vec<serde_json::Value> {
160        self.jobs
161            .lock()
162            .await
163            .values()
164            .map(|job| {
165                serde_json::json!({
166                    "job_id": job.id,
167                    "status": job.status,
168                    "envelope": job.envelope,
169                })
170            })
171            .collect()
172    }
173
174    /// Request cancellation. Returns whether the job existed.
175    ///
176    /// Best-effort by nature: a job that has already finished is reported as
177    /// cancelled-successfully because the caller's intent — "do not keep running
178    /// this" — is already satisfied.
179    ///
180    /// A job discovered `Interrupted` at startup is a different case: its
181    /// `CancellationToken` is fresh and wired to nothing (see
182    /// [`Job::interrupted`]), so calling `.cancel()` on it would silently do
183    /// nothing and leave `status` stuck at `Interrupted` forever, while still
184    /// reporting success — unlike an already-terminal job, its final
185    /// disposition is genuinely unresolved (it could still be resumed), so
186    /// "intent already satisfied" doesn't apply. Instead it's transitioned
187    /// directly to `Cancelled` here. That conveniently also closes the loop
188    /// for resume: a resume check against `status == Interrupted` correctly
189    /// refuses a job cancelled this way, with no extra plumbing needed.
190    ///
191    /// The on-disk ledger is also updated to match: without this, its
192    /// `job_status` table would still say `'running'`, and a subsequent
193    /// daemon restart's startup scan would re-read that, silently re-mark
194    /// the job `Interrupted`, and discard this cancel decision entirely —
195    /// the ledger is the authoritative durability record, so it must not be
196    /// allowed to diverge from the in-memory view like that.
197    pub async fn cancel(&self, id: &str) -> bool {
198        let mut jobs = self.jobs.lock().await;
199        match jobs.get_mut(id) {
200            Some(job) if job.can_resume() => {
201                job.status = JobStatus::Cancelled;
202                job.envelope = None;
203                // Persist to the ledger too — otherwise a restart before
204                // this job's final disposition is otherwise resolved would
205                // silently re-mark it Interrupted, discarding this cancel
206                // decision. Best effort: if the ledger write fails, the
207                // in-memory transition above still holds for this process's
208                // lifetime; only a subsequent restart would be affected,
209                // which is a narrower, more acceptable failure window than
210                // not trying at all.
211                if let Some(jobs_root) = cuttlefish_host::ledger::jobs_root() {
212                    let ledger_path = jobs_root.join(id).join("ledger.sqlite");
213                    // Any existing fingerprint is preserved by
214                    // `Ledger::open` (only inserted on first creation) — the
215                    // value passed here is irrelevant for an
216                    // already-existing ledger, which this always is by the
217                    // time a job can be cancelled.
218                    if let Ok(ledger) = cuttlefish_host::ledger::Ledger::open(&ledger_path, "") {
219                        let _ = ledger.finish("cancelled");
220                    }
221                }
222                true
223            }
224            Some(job) => {
225                job.cancel.cancel();
226                true
227            }
228            None => false,
229        }
230    }
231
232    /// Atomically transition a resumable job to `Running`, if and
233    /// only if [`Job::can_resume`] still holds at this moment. This is the guard
234    /// against two concurrent `POST /jobs/{id}/resume` calls both winning and
235    /// spawning duplicate `run_job` executions against the same ledger.
236    ///
237    /// Returns the job (with its now-updated status) if this call won the
238    /// race, or `None` if another call already claimed it first (or the job
239    /// doesn't exist). Callers must run every other pre-flight check first —
240    /// this method exists to be the *last* thing before `tokio::spawn`, not a
241    /// replacement for the earlier checks; see `resume_job`'s ordering note.
242    pub async fn try_start_resume(&self, id: &str) -> Option<Job> {
243        let mut jobs = self.jobs.lock().await;
244        match jobs.get_mut(id) {
245            Some(job) if job.can_resume() => {
246                job.status = JobStatus::Running;
247                job.cancel = CancellationToken::new();
248                job.envelope = None;
249                // A new subscriber must not mistake the prior attempt's
250                // terminal event for this recovery attempt's result.
251                job.log.clear();
252                Some(job.clone())
253            }
254            _ => None,
255        }
256    }
257}
