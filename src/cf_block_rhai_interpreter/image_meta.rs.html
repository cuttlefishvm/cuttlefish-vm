1//! Image metadata, parsed in the guest without decoding a single pixel.
2//!
3//! Dimensions, EXIF, PNG chunks, JPEG segments. All of it lives in headers,
4//! so all of it is reachable by the same cheap byte parsing as
5//! [`crate::binary`] — no codec, no host round trip, no feature flag.
6//!
7//! That boundary is deliberate and it is where this module stops. Reading
8//! *what an image says about itself* is header parsing. Reading *what the
9//! image looks like* requires decoding, which costs megabytes of codec and
10//! belongs on the host behind a feature flag, the way `pdf-render` already
11//! is. A surprising amount of real triage never crosses that line: when was
12//! this taken, on what device, at what size, has it been re-encoded, does
13//! the EXIF thumbnail still match the claimed content.
14//!
15//! Every parser here assumes hostile input, for the reasons in
16//! [`crate::binary`]'s module docs: a panic in a guest is a wasm trap, and a
17//! trap kills the whole fan-out item instead of failing one file.
18

21/// Pixel dimensions, read from whichever header the format keeps them in.
22///
23/// Cheap enough to run on every file in a corpus, and it is the single most
24/// useful fact about an image after its format — a "photo" that is 1x1 is a
25/// tracking pixel, and one whose dimensions disagree with its EXIF has been
26/// re-encoded or tampered with.
27pub fn dimensions(bytes: &[u8]) -> Result<Value, String> {
28    if bytes.starts_with(b"\x89PNG\r\n\x1a\n") {
29        // IHDR is required to be the first chunk: 8 magic + 4 length + 4 type.
30        let w = be_u32(bytes, 16).ok_or("truncated PNG header")?;
31        let h = be_u32(bytes, 20).ok_or("truncated PNG header")?;
32        return Ok(json!({ "width": w, "height": h, "format": "png" }));
33    }
34    if bytes.starts_with(b"GIF87a") || bytes.starts_with(b"GIF89a") {
35        let w = le_u16(bytes, 6).ok_or("truncated GIF header")?;
36        let h = le_u16(bytes, 8).ok_or("truncated GIF header")?;
37        return Ok(json!({ "width": w, "height": h, "format": "gif" }));
38    }
39    if bytes.starts_with(b"RIFF") && bytes.len() > 12 && &bytes[8..12] == b"WEBP" {
40        // Only the simple lossy form keeps dimensions at a fixed offset;
41        // say so rather than guessing for VP8L/VP8X.
42        if bytes.len() > 30 && &bytes[12..16] == b"VP8 " {
43            let w = le_u16(bytes, 26).ok_or("truncated WebP header")? & 0x3fff;
44            let h = le_u16(bytes, 28).ok_or("truncated WebP header")? & 0x3fff;
45            return Ok(json!({ "width": w, "height": h, "format": "webp" }));
46        }
47        return Err("WebP variant whose dimensions need a decoder".to_string());
48    }
49    if bytes.starts_with(b"\xff\xd8\xff") {
50        return jpeg_dimensions(bytes);
51    }
52    Err("dimensions: unrecognised image format".to_string())
53}

55/// Walk JPEG segments to the frame header that carries the size.
56///
57/// JPEG has no fixed offset for this: the SOFn marker sits after a variable
58/// number of variable-length segments, so it must be walked. Which is also
59/// why this is the parser most likely to meet a lying length field.
60fn jpeg_dimensions(bytes: &[u8]) -> Result<Value, String> {
61    for seg in jpeg_segments(bytes)? {
62        let marker = seg["marker"].as_u64().unwrap_or(0) as u8;
63        // SOF0..SOF15, excluding DHT (0xc4), JPG (0xc8) and DAC (0xcc),
64        // which share the range but are not frame headers.
65        if (0xc0..=0xcf).contains(&marker) && !matches!(marker, 0xc4 | 0xc8 | 0xcc) {
66            let at = seg["offset"].as_u64().unwrap_or(0) as usize;
67            // offset points at the payload: precision(1), height(2), width(2)
68            let h = be_u16(bytes, at + 1).ok_or("truncated JPEG frame header")?;
69            let w = be_u16(bytes, at + 3).ok_or("truncated JPEG frame header")?;
70            return Ok(json!({ "width": w, "height": h, "format": "jpeg" }));
71        }
72    }
73    Err("no JPEG frame header found".to_string())
74}

76/// Every PNG chunk: type, length, offset.
77///
78/// Structure alone answers real questions — a `tEXt`/`iTXt` chunk carrying
79/// a comment, an `eXIf` chunk, or trailing bytes after `IEND`, which is one
80/// of the oldest ways to hide a payload inside an image that still renders.
81pub fn png_chunks(bytes: &[u8]) -> Result<Vec<Value>, String> {
82    if !bytes.starts_with(b"\x89PNG\r\n\x1a\n") {
83        return Err("not a PNG".to_string());
84    }
85    let mut out = Vec::new();
86    let mut at = 8usize;
87    while at + 8 <= bytes.len() {
88        let len = be_u32(bytes, at).ok_or("truncated PNG chunk header")? as usize;
89        let kind = bytes
90            .get(at + 4..at + 8)
91            .ok_or("truncated PNG chunk type")?
92            .to_vec();
93        let name = String::from_utf8_lossy(&kind).to_string();
94        out.push(json!({ "type": name, "length": len, "offset": at }));
95
96        if name == "IEND" {
97            // Anything past IEND is not part of the image. Report it: this
98            // is where appended payloads hide.
99            let end = at + 12;
100            if bytes.len() > end {
101                out.push(json!({
102                    "type": "TRAILING",
103                    "length": bytes.len() - end,
104                    "offset": end,
105                }));
106            }
107            break;
108        }
109        // length + type + data + crc, with checked arithmetic because the
110        // length field is attacker-controlled.
111        at = at
112            .checked_add(12)
113            .and_then(|a| a.checked_add(len))
114            .ok_or("PNG chunk length overflows")?;
115    }
116    Ok(out)
117}

119/// Every JPEG segment: marker, length, payload offset.
120pub fn jpeg_segments(bytes: &[u8]) -> Result<Vec<Value>, String> {
121    if !bytes.starts_with(b"\xff\xd8") {
122        return Err("not a JPEG".to_string());
123    }
124    let mut out = Vec::new();
125    let mut at = 2usize;
126    while at + 4 <= bytes.len() {
127        if bytes[at] != 0xff {
128            // Not at a marker boundary: entropy-coded data, or corruption.
129            break;
130        }
131        let marker = bytes[at + 1];
132        // Start-of-scan is followed by compressed data of unstated length;
133        // walking further would be guessing.
134        if marker == 0xda {
135            out.push(json!({ "marker": marker, "name": "SOS", "offset": at + 2, "length": 0 }));
136            break;
137        }
138        let len = be_u16(bytes, at + 2).ok_or("truncated JPEG segment length")? as usize;
139        if len < 2 {
140            return Err(format!("implausible JPEG segment length {len} at {at}"));
141        }
142        out.push(json!({
143            "marker": marker,
144            "name": jpeg_marker_name(marker),
145            "offset": at + 4,
146            "length": len - 2,
147        }));
148        at = at
149            .checked_add(2)
150            .and_then(|a| a.checked_add(len))
151            .ok_or("JPEG segment length overflows")?;
152    }
153    Ok(out)
154}

156fn jpeg_marker_name(marker: u8) -> &'static str {
157    match marker {
158        0xc0 => "SOF0",
159        0xc2 => "SOF2",
160        0xc4 => "DHT",
161        0xdb => "DQT",
162        0xe0 => "APP0",
163        0xe1 => "APP1",
164        0xed => "APP13",
165        0xee => "APP14",
166        0xfe => "COM",
167        _ => "other",
168    }
169}

171/// EXIF tags from a JPEG's APP1 segment or a raw TIFF header.
172///
173/// Returns only tags worth a triage decision — camera, timestamps, GPS,
174/// orientation, software — rather than the full tag space, because the
175/// consumer is a model or a person deciding what to do next, and a hundred
176/// obscure tags buries the five that matter.
177pub fn exif(bytes: &[u8]) -> Result<Value, String> {
178    let tiff = locate_tiff(bytes)?;
179    let data = &bytes[tiff..];
180
181    let little = match data.get(0..2) {
182        Some(b"II") => true,
183        Some(b"MM") => false,
184        _ => return Err("EXIF has no byte-order mark".to_string()),
185    };
186    let ifd0 = read_u32(data, 4, little).ok_or("truncated EXIF header")? as usize;
187
188    let mut found = serde_json::Map::new();
189    // IFD0, then the Exif sub-IFD it points at, then GPS. Bounded rather
190    // than followed recursively: a crafted file can point an IFD at itself.
191    let mut queue = vec![ifd0];
192    let mut seen = Vec::new();
193    while let Some(offset) = queue.pop() {
194        if seen.contains(&offset) || seen.len() > 8 {
195            continue;
196        }
197        seen.push(offset);
198        for (tag, value) in read_ifd(data, offset, little)? {
199            match tag {
200                // Pointers to further IFDs.
201                0x8769 | 0x8825 => {
202                    if let Some(next) = value.as_u64() {
203                        queue.push(next as usize);
204                    }
205                }
206                0x010f => insert(&mut found, "make", value),
207                0x0110 => insert(&mut found, "model", value),
208                0x0112 => insert(&mut found, "orientation", value),
209                0x0131 => insert(&mut found, "software", value),
210                0x0132 => insert(&mut found, "datetime", value),
211                0x9003 => insert(&mut found, "datetime_original", value),
212                0x829a => insert(&mut found, "exposure_time", value),
213                0x8827 => insert(&mut found, "iso", value),
214                0x0001 => insert(&mut found, "gps_lat_ref", value),
215                0x0003 => insert(&mut found, "gps_lon_ref", value),
216                _ => {}
217            }
218        }
219    }
220    Ok(Value::Object(found))
221}

223fn insert(map: &mut serde_json::Map<String, Value>, key: &str, value: Value) {
224    map.entry(key.to_string()).or_insert(value);
225}

227/// Find the TIFF header EXIF is written inside, in a JPEG or a bare TIFF.
228fn locate_tiff(bytes: &[u8]) -> Result<usize, String> {
229    if bytes.starts_with(b"II*\x00") || bytes.starts_with(b"MM\x00*") {
230        return Ok(0);
231    }
232    if bytes.starts_with(b"\xff\xd8") {
233        for seg in jpeg_segments(bytes)? {
234            if seg["marker"].as_u64() == Some(0xe1) {
235                let at = seg["offset"].as_u64().unwrap_or(0) as usize;
236                if bytes.get(at..at + 6) == Some(b"Exif\x00\x00") {
237                    return Ok(at + 6);
238                }
239            }
240        }
241        return Err("JPEG has no EXIF segment".to_string());
242    }
243    Err("no EXIF: not a JPEG or TIFF".to_string())
244}

246/// Read one IFD's entries as `(tag, value)`, resolving short strings and
247/// integers. Values needing more than four bytes are stored out-of-line.
248fn read_ifd(data: &[u8], offset: usize, little: bool) -> Result<Vec<(u16, Value)>, String> {
249    let count = read_u16(data, offset, little).ok_or("truncated IFD")? as usize;
250    // A crafted count would otherwise drive a very long loop over garbage.
251    if count > 512 {
252        return Err(format!("implausible IFD entry count {count}"));
253    }
254    let mut out = Vec::new();
255    for i in 0..count {
256        let at = offset + 2 + i * 12;
257        let Some(tag) = read_u16(data, at, little) else {
258            break;
259        };
260        let format = read_u16(data, at + 2, little).unwrap_or(0);
261        let components = read_u32(data, at + 4, little).unwrap_or(0) as usize;
262
263        let value = match format {
264            // ASCII
265            2 => {
266                let len = components.saturating_sub(1).min(4096);
267                let start = if components <= 4 {
268                    at + 8
269                } else {
270                    read_u32(data, at + 8, little).unwrap_or(0) as usize
271                };
272                match data.get(start..start + len) {
273                    Some(s) => Value::String(String::from_utf8_lossy(s).trim().to_string()),
274                    None => continue,
275                }
276            }
277            // SHORT
278            3 => match read_u16(data, at + 8, little) {
279                Some(v) => json!(v),
280                None => continue,
281            },
282            // LONG
283            4 => match read_u32(data, at + 8, little) {
284                Some(v) => json!(v),
285                None => continue,
286            },
287            _ => continue,
288        };
289        out.push((tag, value));
290    }
291    Ok(out)
292}

294fn read_u16(d: &[u8], at: usize, little: bool) -> Option<u16> {
295    let b: [u8; 2] = d.get(at..at + 2)?.try_into().ok()?;
296    Some(if little {
297        u16::from_le_bytes(b)
298    } else {
299        u16::from_be_bytes(b)
300    })
301}
302
303fn read_u32(d: &[u8], at: usize, little: bool) -> Option<u32> {
304    let b: [u8; 4] = d.get(at..at + 4)?.try_into().ok()?;
305    Some(if little {
306        u32::from_le_bytes(b)
307    } else {
308        u32::from_be_bytes(b)
309    })
310}
311
312fn be_u16(d: &[u8], at: usize) -> Option<u16> {
313    Some(u16::from_be_bytes(d.get(at..at + 2)?.try_into().ok()?))
314}
315fn be_u32(d: &[u8], at: usize) -> Option<u32> {
316    Some(u32::from_be_bytes(d.get(at..at + 4)?.try_into().ok()?))
317}
318fn le_u16(d: &[u8], at: usize) -> Option<u16> {
319    Some(u16::from_le_bytes(d.get(at..at + 2)?.try_into().ok()?))
320}

391        // A guest panic is a wasm trap that kills the whole fan-out item,
392        // so malformed input must always come back as an error.

412    #[test]
413    fn exif_reads_camera_and_timestamp_from_a_real_tiff_header() {
414        // Little-endian TIFF, one IFD, two ASCII tags stored out-of-line.
415        let mut t = b"II*\x00".to_vec();
416        t.extend_from_slice(&8u32.to_le_bytes()); // IFD0 at 8
417        t.extend_from_slice(&2u16.to_le_bytes()); // 2 entries
418
419        let make = "Canon\0";
420        let dt = "2026:08:09 12:00:00\0";
421        let make_at = 8 + 2 + 24 + 4;
422        let dt_at = make_at + make.len();
423
424        for (tag, len, at) in [(0x010fu16, make.len(), make_at), (0x0132, dt.len(), dt_at)] {
425            t.extend_from_slice(&tag.to_le_bytes());
426            t.extend_from_slice(&2u16.to_le_bytes()); // ASCII
427            t.extend_from_slice(&(len as u32).to_le_bytes());
428            t.extend_from_slice(&(at as u32).to_le_bytes());
429        }
430        t.extend_from_slice(&0u32.to_le_bytes()); // next IFD: none
431        t.extend_from_slice(make.as_bytes());
432        t.extend_from_slice(dt.as_bytes());
433
434        let e = exif(&t).unwrap();
435        assert_eq!(e["make"], "Canon");
436        assert_eq!(e["datetime"], "2026:08:09 12:00:00");
437    }
