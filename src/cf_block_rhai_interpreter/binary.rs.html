1//! Binary inspection for scripts: identify, measure, and parse bytes.
2//!
3//! Everything here is a *pure function over bytes*. A script gets bytes from
4//! `slice_bytes(handle, off, len)`, which returns `{bytes_base64, ...}`, and
5//! passes that string straight in. Nothing in this module talks to the host,
6//! which is what keeps it out of the suspend/replay machinery entirely: a
7//! pure call has no round trip to memoize, so it needs no call index, no
8//! pending command, and no log entry.
9//!
10//! # Why parsing and not decoding
11//!
12//! Reading a container's structure is small — an MP4 atom is a length and a
13//! four-byte tag, a tar header is 512 fixed bytes, a ZIP central directory is
14//! a table. Decoding the *content* is what costs megabytes (video codecs,
15//! image formats), and that belongs on the host behind a feature flag, the
16//! way `pdf-render` already is. This module deliberately stops at structure,
17//! which is where most of the analytical value sits anyway: "what is this,
18//! what's inside it, does it look packed" rarely requires decoding a single
19//! frame.
20//!
21//! The one exception is DEFLATE, which archives need to be useful at all and
22//! which is small enough to carry.
23//!
24//! # Hostile input is the normal case
25//!
26//! These parsers exist to look at files nobody vouched for. A length field
27//! that claims four gigabytes, an atom nested inside itself, a truncated
28//! header — all are ordinary here, not edge cases. Every parser is written
29//! to return an error rather than panic, because a panic in a guest is a
30//! wasm trap, and a trap kills the whole fan-out item rather than recording
31//! "this one file is malformed" and moving on.
32

36/// Decode the base64 a script got from `slice_bytes`.
37pub fn decode(bytes_base64: &str) -> Result<Vec<u8>, String> {
38    base64::engine::general_purpose::STANDARD
39        .decode(bytes_base64.trim())
40        .map_err(|e| format!("not valid base64: {e}"))
41}

43/// Shannon entropy in bits per byte, 0.0 to 8.0.
44///
45/// The cheapest signal there is for "are these bytes compressed, encrypted,
46/// or packed": English text sits near 4.5, a PNG or a zip near 7.99. It
47/// cannot distinguish encryption from compression — both are near-maximal —
48/// so it answers "is this opaque", not "is this a secret".
49pub fn entropy(bytes: &[u8]) -> f64 {
50    if bytes.is_empty() {
51        return 0.0;
52    }
53    let mut counts = [0u64; 256];
54    for &b in bytes {
55        counts[b as usize] += 1;
56    }
57    let len = bytes.len() as f64;
58    -counts
59        .iter()
60        .filter(|&&c| c > 0)
61        .map(|&c| {
62            let p = c as f64 / len;
63            p * p.log2()
64        })
65        .sum::<f64>()
66}

68/// How often each byte value occurs, as a 256-element array.
69pub fn byte_histogram(bytes: &[u8]) -> Vec<u64> {
70    let mut counts = vec![0u64; 256];
71    for &b in bytes {
72        counts[b as usize] += 1;
73    }
74    counts
75}

77/// Runs of printable ASCII at least `min_len` long, with their offsets.
78///
79/// The first thing anyone runs on an unknown binary. Printable is taken as
80/// 0x20..=0x7e plus tab — deliberately not including newline, so a run
81/// reported here is one line's worth of text rather than a whole file
82/// collapsed into a single entry.
83pub fn strings(bytes: &[u8], min_len: usize) -> Vec<Value> {
84    let min_len = min_len.max(1);
85    let mut out = Vec::new();
86    let mut start = 0usize;
87    let mut run = 0usize;
88
89    let flush = |start: usize, run: usize, out: &mut Vec<Value>| {
90        if run >= min_len {
91            let text = String::from_utf8_lossy(&bytes[start..start + run]).into_owned();
92            out.push(json!({ "offset": start, "text": text }));
93        }
94    };
95
96    for (i, &b) in bytes.iter().enumerate() {
97        let printable = (0x20..=0x7e).contains(&b) || b == b'\t';
98        if printable {
99            if run == 0 {
100                start = i;
101            }
102            run += 1;
103        } else {
104            flush(start, run, &mut out);
105            run = 0;
106        }
107    }
108    flush(start, run, &mut out);
109    out
110}

112/// Classic hex+ASCII dump, 16 bytes per line.
113///
114/// Exists so a *model* can read the bytes. Handing a vision-less model raw
115/// base64 tells it nothing; a hexdump is the representation everything in
116/// its training data uses for "here are some bytes".
117pub fn hexdump(bytes: &[u8], base_offset: u64) -> String {
118    let mut out = String::new();
119    for (row, chunk) in bytes.chunks(16).enumerate() {
120        let offset = base_offset + (row * 16) as u64;
121        out.push_str(&format!("{offset:08x}  "));
122        for i in 0..16 {
123            match chunk.get(i) {
124                Some(b) => out.push_str(&format!("{b:02x} ")),
125                None => out.push_str("   "),
126            }
127            if i == 7 {
128                out.push(' ');
129            }
130        }
131        out.push_str(" |");
132        for &b in chunk {
133            out.push(if (0x20..=0x7e).contains(&b) {
134                b as char
135            } else {
136                '.'
137            });
138        }
139        out.push_str("|\n");
140    }
141    out
142}

144/// A format guess from leading bytes.
145///
146/// Content, never the file name — the whole point is to be right about a
147/// file called `data.txt` that is really a zip. `confidence` is coarse on
148/// purpose: `high` for a magic number long and specific enough that a false
149/// positive is implausible, `medium` for short or common signatures.
150pub fn identify(bytes: &[u8]) -> Value {
151    // Ordered longest/most-specific first, since some signatures are
152    // prefixes of others and the first match wins.
153    const SIGNATURES: &[(&[u8], &str, &str, &str)] = &[
154        (b"\x89PNG\r\n\x1a\n", "png", "image/png", "high"),
155        (b"GIF89a", "gif", "image/gif", "high"),
156        (b"GIF87a", "gif", "image/gif", "high"),
157        (b"%PDF-", "pdf", "application/pdf", "high"),
158        (b"\x7fELF", "elf", "application/x-executable", "high"),
159        (b"PK\x03\x04", "zip", "application/zip", "high"),
160        (b"PK\x05\x06", "zip", "application/zip", "high"),
161        (b"\x1f\x8b", "gzip", "application/gzip", "high"),
162        (b"BZh", "bzip2", "application/x-bzip2", "medium"),
163        (b"\xfd7zXZ\x00", "xz", "application/x-xz", "high"),
164        (b"\x28\xb5\x2f\xfd", "zstd", "application/zstd", "high"),
165        (
166            b"7z\xbc\xaf\x27\x1c",
167            "7z",
168            "application/x-7z-compressed",
169            "high",
170        ),
171        (b"Rar!\x1a\x07", "rar", "application/vnd.rar", "high"),
172        (b"OggS", "ogg", "application/ogg", "high"),
173        (b"fLaC", "flac", "audio/flac", "high"),
174        (b"\xff\xd8\xff", "jpeg", "image/jpeg", "high"),
175        (
176            b"MZ",
177            "pe",
178            "application/vnd.microsoft.portable-executable",
179            "medium",
180        ),
181        (
182            b"\x25\x21PS",
183            "postscript",
184            "application/postscript",
185            "medium",
186        ),
187        (
188            b"SQLite format 3\x00",
189            "sqlite",
190            "application/vnd.sqlite3",
191            "high",
192        ),
193        (b"\x00asm", "wasm", "application/wasm", "high"),
194    ];
195
196    for (magic, format, mime, confidence) in SIGNATURES {
197        if bytes.starts_with(magic) {
198            return json!({ "format": format, "mime": mime, "confidence": confidence });
199        }
200    }
201
202    // Signatures that aren't at offset 0.
203    if bytes.len() >= 12 && &bytes[4..8] == b"ftyp" {
204        // The brand distinguishes MP4 from HEIF and friends; report it
205        // rather than flattening every ISO-BMFF file to "mp4".
206        let brand = String::from_utf8_lossy(&bytes[8..12]).trim().to_string();
207        let (format, mime) = match &brand[..brand.len().min(3)] {
208            "hei" | "mif" => ("heif", "image/heif"),
209            "qt " => ("mov", "video/quicktime"),
210            _ => ("mp4", "video/mp4"),
211        };
212        return json!({ "format": format, "mime": mime, "confidence": "high", "brand": brand });
213    }
214    if bytes.len() >= 12 && bytes.starts_with(b"RIFF") {
215        let form = String::from_utf8_lossy(&bytes[8..12]).to_string();
216        let (format, mime) = match form.as_str() {
217            "WAVE" => ("wav", "audio/wav"),
218            "AVI " => ("avi", "video/x-msvideo"),
219            "WEBP" => ("webp", "image/webp"),
220            _ => ("riff", "application/octet-stream"),
221        };
222        return json!({ "format": format, "mime": mime, "confidence": "high" });
223    }
224    // A tar has no leading magic at all — its signature sits at offset 257,
225    // which is why a tarball is so often misidentified as "data".
226    if bytes.len() >= 262 && (&bytes[257..262] == b"ustar") {
227        return json!({ "format": "tar", "mime": "application/x-tar", "confidence": "high" });
228    }
229
230    if std::str::from_utf8(bytes).is_ok() {
231        return json!({ "format": "text", "mime": "text/plain", "confidence": "medium" });
232    }
233    json!({ "format": "unknown", "mime": "application/octet-stream", "confidence": "low" })
234}

236#[cfg(test)]

240    #[test]
241    fn entropy_separates_uniform_from_random_looking_bytes() {
242        // All one byte: no information at all.
243        assert_eq!(entropy(&[0u8; 1024]), 0.0);
244        // Every byte value once: maximal.
245        let all: Vec<u8> = (0..=255).collect();
246        assert!((entropy(&all) - 8.0).abs() < 1e-9, "{}", entropy(&all));
247        // Empty is defined, not a divide-by-zero.
248        assert_eq!(entropy(&[]), 0.0);
249        // Ordinary English sits well below the opaque threshold, which is
250        // the discrimination the function exists to provide.
251        assert!(entropy(b"the quick brown fox jumps over the lazy dog") < 5.0);
252    }
253
254    #[test]
255    fn strings_finds_runs_and_respects_the_minimum() {
256        let data = b"\x00\x01hello\x00ab\x00world!\xff";
257        let found = strings(data, 5);
258        let texts: Vec<String> = found
259            .iter()
260            .map(|v| v["text"].as_str().unwrap().to_string())
261            .collect();
262        assert_eq!(
263            texts,
264            vec!["hello", "world!"],
265            "short run `ab` must be dropped"
266        );
267        // Offsets must be real, or a caller can't seek back to what it found.
268        assert_eq!(found[0]["offset"], 2);
269    }

271    #[test]
272    fn strings_terminates_a_run_at_the_end_of_input() {
273        // The off-by-one that a loop-only implementation misses: a run
274        // touching the last byte is never flushed unless the end is handled.
275        let found = strings(b"\x00trailing", 4);
276        assert_eq!(found.len(), 1);
277        assert_eq!(found[0]["text"], "trailing");
278    }

280    #[test]
281    fn identify_reads_content_not_names() {
282        assert_eq!(identify(b"\x89PNG\r\n\x1a\n\x00\x00")["format"], "png");
283        assert_eq!(identify(b"PK\x03\x04rest")["format"], "zip");
284        assert_eq!(identify(b"\x00asm\x01\x00\x00\x00")["format"], "wasm");
285        assert_eq!(identify(b"plain words here")["format"], "text");
286        assert_eq!(identify(&[0x00, 0xff, 0xfe, 0x01])["format"], "unknown");
287    }

289    #[test]
290    fn identify_finds_signatures_that_are_not_at_offset_zero() {
291        // A tar's `ustar` marker lives at 257, which is why tarballs so
292        // often get reported as featureless data.
293        let mut tar = vec![0u8; 512];
294        tar[257..262].copy_from_slice(b"ustar");
295        assert_eq!(identify(&tar)["format"], "tar");
296
297        // ISO-BMFF: the brand at offset 8 separates mp4 from heif/mov.
298        let mut mp4 = vec![0u8; 16];
299        mp4[4..8].copy_from_slice(b"ftyp");
300        mp4[8..12].copy_from_slice(b"isom");
301        assert_eq!(identify(&mp4)["format"], "mp4");
302        mp4[8..12].copy_from_slice(b"heic");
303        assert_eq!(identify(&mp4)["format"], "heif");
304
305        let mut wav = b"RIFF\x00\x00\x00\x00WAVE".to_vec();
306        wav.extend_from_slice(&[0u8; 4]);
307        assert_eq!(identify(&wav)["format"], "wav");
308    }

310    #[test]
311    fn identify_never_panics_on_short_or_hostile_input() {
312        // Every length from empty to past the tar marker: the parsers index
313        // at fixed offsets, and a guest panic is a wasm trap that kills the
314        // whole fan-out item rather than failing one file.
315        for len in 0..300 {
316            let _ = identify(&vec![0xffu8; len]);
317            let _ = identify(&vec![0x00u8; len]);
318        }

321    #[test]
322    fn hexdump_pads_a_short_final_row_and_stays_aligned() {
323        let dump = hexdump(b"AB", 0);
324        assert!(dump.starts_with("00000000  41 42 "), "{dump}");
325        assert!(dump.trim_end().ends_with("|AB|"), "{dump}");
326        // Unprintables must render as dots, not as raw control characters
327        // that would corrupt the surrounding output.
328        assert!(hexdump(&[0x00, 0x07], 0).trim_end().ends_with("|..|"));
329    }

331    #[test]
332    fn hexdump_offsets_are_absolute_not_relative() {
333        // A caller dumping a window mid-file needs offsets it can seek to.
334        let dump = hexdump(&[0u8; 17], 0x1000);
335        assert!(dump.starts_with("00001000  "), "{dump}");
336        assert!(dump.contains("\n00001010  "), "{dump}");
337    }
