1//! Reads a spreadsheet into JSON rows, entirely in the guest.
2//!
3//! XLSX, XLSM and XLS open as `kind: "binary"` with zero characters through
4//! the text path, because they are not text — an XLSX is a zip of XML, and
5//! an XLS is a compound binary document. A corpus where the numeric outcome
6//! series live in workbooks is therefore unreadable by any amount of
7//! `slice`/`page_text`, which is exactly the situation the Medicaid 1115
8//! field log describes: 1,059 of 8,185 files, and not incidental ones.
9//!
10//! # Why this is a Rust block and not a Rhai builtin
11//!
12//! The pattern established by the binary and image toolkits is that *header
13//! parsing* belongs in the shared interpreter, because it is small, and
14//! *decoding* belongs elsewhere, because it is not. A sheet reader is
15//! decoding: `calamine` brings a zip reader, an XML parser, a shared-string
16//! table, a date serial converter and the compound-file reader for legacy
17//! XLS. Putting that in the interpreter would tax every scripted job in the
18//! system for a capability most of them never use.
19//!
20//! It stays in the guest rather than becoming a host feature flag for a
21//! different reason: `calamine` compiles to `wasm32-unknown-unknown` with
22//! default features off, so nothing here needs native code. A capability
23//! that needs no feature flag is one nobody has to discover they lack.
24//!
25//! # Memory
26//!
27//! The whole workbook is pulled into guest memory before parsing, because
28//! `calamine` needs `Read + Seek` over the complete file — a zip's central
29//! directory is at the end. That is a real ceiling and the reason
30//! `MAX_BYTES` exists: a workbook past it fails as one item with a clear
31//! message rather than trapping the guest and taking the whole fan-out item
32//! with it.
33
34use cuttlefish_sdk::{export_block, Block, Command, Event, MediaKind, Signature};
35
36/// Largest workbook this block will hold in guest memory.
37///
38/// Chosen against the corpus that motivated the block: CMS budget-neutrality
39/// workbooks run to a few megabytes, so 64 MiB is generous by roughly an
40/// order of magnitude while still bounding the failure. Exceeding it is an
41/// ordinary item failure, not a trap.
42const MAX_BYTES: u64 = 64 * 1024 * 1024;
43
44/// How much to ask for per `SliceBytes`.
45///
46/// The host base64-encodes each window, so a very large window costs a third
47/// again in transfer for no benefit. A megabyte keeps the round-trip count
48/// low without making any single message enormous.
49const WINDOW: u64 = 1024 * 1024;
50
51#[derive(Default)]
52struct SheetExtract {
53    path: String,
54    handle: u32,
55    len: u64,
56    /// The workbook, accumulated across windows.
57    bytes: Vec<u8>,
58    /// How many rows of each sheet to return, from the job input.
59    max_rows: usize,
60}
61
62/// Rows a sheet returns when the caller does not say.
63///
64/// Not unbounded: a single sheet in this corpus can carry tens of thousands
65/// of rows, and a fan-out item returning all of them puts the whole thing
66/// through the ledger and into `results.jsonl`. A caller that wants
67/// everything can ask for it explicitly.
68const DEFAULT_MAX_ROWS: usize = 1000;
69
70impl Block for SheetExtract {
71    fn signature() -> Signature {
72        // Stated concretely rather than `json -> json`: a `json` seam
73        // typechecks unconditionally, which is the same as not checking it.
74        Signature {
75            input: "{path: text}".parse().expect("a literal type"),
76            output: "{path: text, sheets: json, schema: json, truncated: json}"
77                .parse()
78                .expect("a literal type"),
79        }
80    }
81
82    fn start(&mut self, input: serde_json::Value) -> Command {
83        self.max_rows = input
84            .get("max_rows")
85            .and_then(|v| v.as_u64())
86            .map(|n| n as usize)
87            .unwrap_or(DEFAULT_MAX_ROWS);
88
89        match input.get("path").and_then(|v| v.as_str()) {
90            Some(path) => {
91                self.path = path.to_string();
92                Command::Open {
93                    path: path.to_string(),
94                }
95            }
96            // Fail rather than panic: a panic is an opaque wasm trap, while
97            // this reaches the caller as a code they can act on.
98            None => fail(
99                "schema_validation_failed",
100                "input must have a string `path` field",
101            ),
102        }
103    }
104
105    fn step(&mut self, event: Event) -> Command {
106        match event {
107            Event::Opened { handle, len, kind } => {
108                // A workbook is `Binary` to the host, which has no sheet
109                // reader — that is the whole reason this block exists. Text
110                // and documents are somebody else's job, and saying so is
111                // more useful than reading them badly.
112                if !matches!(kind, MediaKind::Binary) {
113                    return fail(
114                        "unsupported",
115                        &format!(
116                            "`{}` opened as {kind:?}, not a binary workbook. This block reads \
117                             XLSX/XLSM/XLS; use `page_text` or `slice` for text and documents.",
118                            self.path
119                        ),
120                    );
121                }
122                if len > MAX_BYTES {
123                    return fail(
124                        "unsupported",
125                        &format!(
126                            "workbook is {len} bytes, over this block's {MAX_BYTES}-byte ceiling. \
127                             The whole file must be in guest memory to be parsed, since a zip's \
128                             central directory is at the end."
129                        ),
130                    );
131                }
132                self.handle = handle;
133                self.len = len;
134                self.bytes = Vec::with_capacity(len as usize);
135                self.next_window()
136            }
137
138            Event::SlicedBytes { bytes_base64, .. } => {
139                use base64::Engine as _;
140                match base64::engine::general_purpose::STANDARD.decode(&bytes_base64) {
141                    Ok(mut chunk) => self.bytes.append(&mut chunk),
142                    Err(e) => return fail("unsupported", &format!("undecodable window: {e}")),
143                }
144                if (self.bytes.len() as u64) < self.len {
145                    return self.next_window();
146                }
147                self.parse()
148            }
149
150            other => fail(
151                "unsupported",
152                &format!("this block issues only Open and SliceBytes; got {other:?}"),
153            ),
154        }
155    }
156}

158impl SheetExtract {
159    fn next_window(&self) -> Command {
160        let offset = self.bytes.len() as u64;
161        Command::SliceBytes {
162            handle: self.handle,
163            offset,
164            len: WINDOW.min(self.len - offset),
165        }
166    }
167
168    /// Parse the assembled bytes and finish.
169    fn parse(&mut self) -> Command {
170        use calamine::Reader;
171
172        let cursor = std::io::Cursor::new(std::mem::take(&mut self.bytes));
173        let mut workbook = match calamine::open_workbook_auto_from_rs(cursor) {
174            Ok(w) => w,
175            // A workbook that will not open is a data-quality fact about one
176            // file, not an authoring error, so it fails this item and lets a
177            // fan-out run continue.
178            Err(e) => {
179                return fail(
180                    "unsupported",
181                    &format!("`{}` is not a readable workbook: {e}", self.path),
182                )
183            }
184        };
185
186        let mut sheets = serde_json::Map::new();
187        let mut schema = serde_json::Map::new();
188        let mut truncated = serde_json::Map::new();
189
190        for name in workbook.sheet_names().to_owned() {
191            let Ok(range) = workbook.worksheet_range(&name) else {
192                // One unreadable sheet does not condemn the workbook: record
193                // it and keep the others, the same way one bad item does not
194                // condemn a fan-out.
195                sheets.insert(name.clone(), serde_json::Value::Null);
196                schema.insert(name.clone(), serde_json::Value::Null);
197                continue;
198            };
199
200            let total = range.rows().count();
201            // Schema is computed over the *whole* range, before truncation.
202            // Inferring a column's type from the first thousand rows of a
203            // fifty-thousand-row sheet is how a column that turns numeric
204            // halfway down gets reported as text.
205            schema.insert(name.clone(), describe(&range));
206
207            let rows: Vec<serde_json::Value> = range
208                .rows()
209                .take(self.max_rows)
210                .map(|row| serde_json::Value::Array(row.iter().map(cell_to_json).collect()))
211                .collect();
212
213            if total > rows.len() {
214                // Reported, never silent. A caller summing a column has to
215                // know the column was cut off, or the total is wrong and
216                // looks right.
217                truncated.insert(
218                    name.clone(),
219                    serde_json::json!({ "returned": rows.len(), "total": total }),
220                );
221            }
222            sheets.insert(name, serde_json::Value::Array(rows));
223        }
224
225        Command::Done {
226            result: serde_json::json!({
227                "path": self.path,
228                "sheets": serde_json::Value::Object(sheets),
229                "schema": serde_json::Value::Object(schema),
230                "truncated": serde_json::Value::Object(truncated),
231            }),
232        }
233    }
234}

236/// One cell as JSON.
237///
238/// Numbers stay numbers and strings stay strings, so a downstream node can do
239/// arithmetic without reparsing. Dates and times are rendered as strings via
240/// calamine's own conversion rather than left as the raw serial number: a
241/// bare `45292.0` in a report is indistinguishable from a quantity, which is
242/// exactly the kind of silently-wrong value this codebase tries not to
243/// produce.
244fn cell_to_json(cell: &calamine::Data) -> serde_json::Value {
245    use calamine::Data;
246    match cell {
247        Data::Empty => serde_json::Value::Null,
248        Data::String(s) => serde_json::Value::String(s.clone()),
249        Data::Float(f) => serde_json::Number::from_f64(*f)
250            .map(serde_json::Value::Number)
251            // NaN and infinity have no JSON representation; a string keeps
252            // the fact rather than dropping the cell.
253            .unwrap_or_else(|| serde_json::Value::String(f.to_string())),
254        Data::Int(i) => serde_json::Value::Number((*i).into()),
255        Data::Bool(b) => serde_json::Value::Bool(*b),
256        Data::DateTime(d) => serde_json::Value::String(
257            d.as_datetime()
258                .map(|dt| dt.to_string())
259                .unwrap_or_else(|| d.as_f64().to_string()),
260        ),
261        Data::DateTimeIso(s) => serde_json::Value::String(s.clone()),
262        Data::DurationIso(s) => serde_json::Value::String(s.clone()),
263        // An error cell (#REF!, #DIV/0!) is a fact worth carrying, not a
264        // blank: a blank reads as "no data", which is a different claim.
265        Data::Error(e) => serde_json::Value::String(format!("#ERROR:{e:?}")),
266    }
267}

272        message: message.into(),

278/// Where the table actually is, and what its columns hold.
279///
280/// Real workbooks rarely put a header at A1. A CMS budget-neutrality
281/// workbook opens with a title, a blank row, an "as of" date, maybe a merged
282/// banner, and only then the column names — and the data may be inset from
283/// column A as well. A caller that assumes row 0 is the header gets a header
284/// of `["Demonstration Budget Neutrality", "", "", ""]` and column names
285/// that are really data.
286///
287/// So this reports the offsets rather than assuming them, and reports the
288/// *evidence* alongside, because the detection is a heuristic and a Rhai
289/// block cleaning this up for a knowledge base needs to be able to disagree
290/// with it. Everything here is descriptive: nothing is dropped or rewritten,
291/// and the rows are still returned verbatim.
292fn describe(range: &calamine::Range<calamine::Data>) -> serde_json::Value {
293    let rows: Vec<&[calamine::Data]> = range.rows().collect();
294    // `start` is where the sheet's used range begins in the real grid —
295    // a table inset from A1 would otherwise report column indices that do
296    // not match the spreadsheet a human is looking at.
297    let (row_offset, col_offset) = range.start().unwrap_or((0, 0));
298
299    let header_row = detect_header_row(&rows);
300    let data_start = header_row.map(|h| h + 1).unwrap_or(0);
301
302    let width = rows.iter().map(|r| r.len()).max().unwrap_or(0);
303    let columns: Vec<serde_json::Value> = (0..width)
304        .map(|c| {
305            let name = header_row
306                .and_then(|h| rows.get(h))
307                .and_then(|r| r.get(c))
308                .map(cell_to_text)
309                .filter(|s| !s.is_empty());
310
311            let cells: Vec<&calamine::Data> = rows
312                .iter()
313                .skip(data_start)
314                .filter_map(|r| r.get(c))
315                .collect();
316            let filled = cells
317                .iter()
318                .filter(|c| !matches!(c, calamine::Data::Empty))
319                .count();
320
321            serde_json::json!({
322                "index": c,
323                // The spreadsheet's own column letter, so a finding can be
324                // pointed at in Excel without arithmetic.
325                "letter": column_letter(col_offset as usize + c),
326                "name": name,
327                "type": infer_type(&cells),
328                "filled": filled,
329                "of": cells.len(),
330                // A sample is worth more than a type name when the type is
331                // "mixed" and somebody has to decide what to do about it.
332                "sample": cells
333                    .iter()
334                    .find(|c| !matches!(c, calamine::Data::Empty))
335                    .map(|c| cell_to_json(c))
336                    .unwrap_or(serde_json::Value::Null),
337            })
338        })
339        .collect();
340
341    serde_json::json!({
342        "header_row": header_row,
343        "data_start_row": data_start,
344        "rows": rows.len(),
345        "columns": columns,
346        // Where the used range sits in the real grid, so reported indices
347        // can be translated back to what a person sees.
348        "grid_origin": { "row": row_offset, "col": col_offset },
349    })
350}

352/// Guess which row holds the column names.
353///
354/// The heuristic: among the first [`HEADER_SEARCH_DEPTH`] rows, prefer the
355/// one with the most non-empty *text* cells that is also followed by a row
356/// of comparable width. Titles and banners lose because they are one wide
357/// cell; a header row loses to nothing because it is many narrow ones.
358///
359/// Returns `None` when no row looks like a header — a sheet that is pure
360/// numbers, or empty. `None` is a real answer and better than pointing at
361/// row 0 and calling it names.
362fn detect_header_row(rows: &[&[calamine::Data]]) -> Option<usize> {
363    const HEADER_SEARCH_DEPTH: usize = 25;
364
365    let mut best: Option<(usize, usize)> = None;
366    for (i, row) in rows.iter().enumerate().take(HEADER_SEARCH_DEPTH) {
367        let texty = row
368            .iter()
369            .filter(|c| matches!(c, calamine::Data::String(s) if !s.trim().is_empty()))
370            .count();
371        if texty < 2 {
372            // One text cell is a title, not a header.
373            continue;
374        }
375        // The row beneath must carry something, or this is a trailing label
376        // block rather than the top of a table.
377        let below_filled = rows
378            .get(i + 1)
379            .map(|r| {
380                r.iter()
381                    .filter(|c| !matches!(c, calamine::Data::Empty))
382                    .count()
383            })
384            .unwrap_or(0);
385        if below_filled == 0 {
386            continue;
387        }
388        if best.is_none_or(|(_, score)| texty > score) {
389            best = Some((i, texty));
390        }
391    }
392    best.map(|(i, _)| i)
393}

395/// What a column holds, across its data rows.
396///
397/// `mixed` is reported rather than resolved. A column of numbers with three
398/// stray "N/A" strings is a fact the caller needs; silently calling it text
399/// (or worse, numeric) is how a total ends up wrong and plausible.
400fn infer_type(cells: &[&calamine::Data]) -> &'static str {
401    use calamine::Data;
402    let (mut num, mut text, mut date, mut boolean, mut err) = (0, 0, 0, 0, 0);
403    for cell in cells {
404        match cell {
405            Data::Empty => {}
406            Data::Int(_) | Data::Float(_) => num += 1,
407            Data::String(_) => text += 1,
408            Data::DateTime(_) | Data::DateTimeIso(_) | Data::DurationIso(_) => date += 1,
409            Data::Bool(_) => boolean += 1,
410            Data::Error(_) => err += 1,
411        }
412    }
413    let present = num + text + date + boolean + err;
414    if present == 0 {
415        return "empty";
416    }
417    match (num, text, date, boolean, err) {
418        (n, 0, 0, 0, 0) if n > 0 => "number",
419        (0, t, 0, 0, 0) if t > 0 => "text",
420        (0, 0, d, 0, 0) if d > 0 => "date",
421        (0, 0, 0, b, 0) if b > 0 => "bool",
422        _ => "mixed",
423    }
424}

426/// Zero-based column index to its spreadsheet letter: 0 -> A, 26 -> AA.
427fn column_letter(mut index: usize) -> String {
428    let mut out = Vec::new();
429    loop {
430        out.push(b'A' + (index % 26) as u8);
431        if index < 26 {
432            break;
433        }
434        index = index / 26 - 1;
435    }
436    out.reverse();
437    String::from_utf8(out).expect("ASCII letters")
438}

440fn cell_to_text(cell: &calamine::Data) -> String {
441    match cell {
442        calamine::Data::String(s) => s.trim().to_string(),
443        calamine::Data::Empty => String::new(),
444        other => cell_to_json(other).to_string(),
445    }
446}

452    #[test]
453    fn column_letters_match_the_spreadsheet() {
454        // So a finding can be pointed at in Excel without arithmetic.
455        assert_eq!(column_letter(0), "A");
456        assert_eq!(column_letter(25), "Z");
457        assert_eq!(column_letter(26), "AA");
458        assert_eq!(column_letter(27), "AB");
459        assert_eq!(column_letter(51), "AZ");
460        assert_eq!(column_letter(52), "BA");
461    }
462
463    #[test]
464    fn a_column_of_numbers_with_stray_text_is_mixed_not_number() {
465        use calamine::Data;
466        let clean = [Data::Float(1.0), Data::Float(2.0), Data::Empty];
467        let cells: Vec<&Data> = clean.iter().collect();
468        assert_eq!(infer_type(&cells), "number");
469
470        // The case that matters: three "N/A"s in a numeric column. Calling
471        // this "number" is how a total comes out wrong and plausible.
472        let dirty = [Data::Float(1.0), Data::String("N/A".into())];
473        let cells: Vec<&Data> = dirty.iter().collect();
474        assert_eq!(infer_type(&cells), "mixed");
475
476        let none: Vec<&Data> = Vec::new();
477        assert_eq!(infer_type(&none), "empty");
478    }

480    #[test]
481    fn a_title_row_does_not_win_over_the_real_header() {
482        use calamine::Data;
483        // Shaped like a real workbook: title, blank, then the header.
484        let r0 = vec![Data::String("Demonstration Budget Neutrality".into())];
485        let r1 = vec![Data::Empty, Data::Empty];
486        let r2 = vec![
487            Data::String("member_months".into()),
488            Data::String("cost".into()),
489        ];
490        let r3 = vec![Data::Float(1200.0), Data::Float(4567.89)];
491        let rows: Vec<&[Data]> = vec![&r0, &r1, &r2, &r3];
492
493        assert_eq!(
494            detect_header_row(&rows),
495            Some(2),
496            "a one-cell title is not a header; the two-name row is"
497        );
498    }
